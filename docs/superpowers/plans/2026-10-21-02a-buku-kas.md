# UDP Tracker — Rencana 2a: Buku Kas Inti

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Pengguna bisa mencatat pengeluaran/pemasukan/transfer (dengan biaya admin), melihat dan memfilter daftar transaksi, melihat beranda keuangan (UDP diterima, pengeluaran, sisa, saldo per dompet, Pagu Makan dan "ditanggung kantor"), serta kalender bulanan. Tap tanggal di kalender membuka detail pengeluaran hari itu.

**Architecture:** Mengikuti `docs/architecture/data-flow.md`. Semua angka dihitung di `lib/` (fungsi murni) dari satu `Ledger` per proyek (`getLedger`: proyek, dompet, kategori, pagu, transaksi aktif). Query beranda/kalender/daftar menurunkan view dari Ledger itu, dan route `/api/projects/[projectId]/*` memakai query yang sama. Mutasi transaksi lewat command → `planTransaction` (zod + aturan §3.4) → repo. Browser meng-invalidate `ledgerKeys.all(projectId)` setelah mutasi.

**Tech Stack:** sama dengan Rencana 1 (Next.js 16.4.2 cacheComponents, Drizzle + pg → Neon, TanStack Query, NeoBrutalism/Base UI, Vitest + PGlite).

**Spec:** `docs/superpowers/specs/2026-10-21-udp-tracker-design.md` §3.3, §3.4, §5 (Beranda, Kalender, Transaksi, Form transaksi).

**Di luar rencana ini:** scan/lampiran bon (Rencana 3); kelola dompet + cek saldo, kategori/pagu/mobil, pengaturan proyek (Rencana 2b); ekspor (Rencana 4).

```ts
import "server-only";
import { and, desc, eq, isNull } from "drizzle-orm";
import { transactions } from "@/db/schema";
import type { Db } from "@/db/types";

export type TransactionRow = typeof transactions.$inferSelect;
export type NewTransactionRow = typeof transactions.$inferInsert;

/** Transaksi aktif (belum dihapus) proyek, terbaru dulu. */
export function selectTransactions(db: Db, projectId: string): Promise<TransactionRow[]> {
  return db
    .select()
    .from(transactions)
    .where(and(eq(transactions.projectId, projectId), isNull(transactions.deletedAt)))
    .orderBy(desc(transactions.date), desc(transactions.createdAt));
}

/** Termasuk yang sudah dihapus (dipakai untuk urungkan). */
export async function selectTransaction(db: Db, projectId: string, id: string): Promise<TransactionRow | undefined> {
  const [row] = await db.select().from(transactions).where(and(eq(transactions.projectId, projectId), eq(transactions.id, id)));
  return row;
}

export async function insertTransaction(db: Db, values: NewTransactionRow): Promise<TransactionRow> {
  const [row] = await db.insert(transactions).values(values).returning();
  return row;
}

export async function updateTransactionRow(db: Db, id: string, patch: Partial<NewTransactionRow>): Promise<TransactionRow> {
  const [row] = await db.update(transactions).set(patch).where(eq(transactions.id, id)).returning();
  return row;
}
```

```ts
// Perhitungan buku kas (spec §3.3). Murni: dipakai beranda, kalender, daftar, dan laporan.
import type { Category } from "@/modules/budgets/types/category";
import type { Vehicle } from "@/modules/projects/types/project";
import { sum } from "@/modules/shared/lib/money";
import type { Wallet } from "@/modules/wallets/types/wallet";
import type { ExpenseEntry, LedgerTotals, Transaction, TransactionView } from "../types/transaction";

/** Semua pengeluaran: transaksi expense + biaya admin transfer (kategori isAdminFee; bukan baris transaksi kedua). */
export function expenseEntries(transactions: Transaction[], categories: Category[]): ExpenseEntry[] {
  const adminCategoryId = categories.find((c) => c.isAdminFee)?.id ?? null;
  const entries: ExpenseEntry[] = [];
  for (const t of transactions) {
    if (t.type === "expense") {
      entries.push({ transactionId: t.id, date: t.date, categoryId: t.categoryId, walletId: t.walletId, amount: t.amount });
    } else if (t.type === "transfer" && t.adminFee > 0) {
      entries.push({ transactionId: t.id, date: t.date, categoryId: adminCategoryId, walletId: t.walletId, amount: t.adminFee });
    }
  }
  return entries;
}

export function ledgerTotals(wallets: Wallet[], categories: Category[], transactions: Transaction[]): LedgerTotals {
  const officeFunding = new Set(categories.filter((c) => c.isOfficeFunding).map((c) => c.id));
  const income = transactions.filter((t) => t.type === "income");
  const openingTotal = sum(wallets.map((w) => w.openingBalance));
  const totalIncome = sum(income.map((t) => t.amount));
  const udpReceived = sum(income.filter((t) => t.categoryId && officeFunding.has(t.categoryId)).map((t) => t.amount));
  const totalExpense = sum(expenseEntries(transactions, categories).map((e) => e.amount));
  return { openingTotal, totalIncome, udpReceived, totalExpense, remaining: openingTotal + totalIncome - totalExpense };
}

export function sumBy<T, K>(items: T[], key: (item: T) => K, value: (item: T) => number): Map<K, number> {
  const totals = new Map<K, number>();
  for (const item of items) totals.set(key(item), (totals.get(key(item)) ?? 0) + value(item));
  return totals;
}

export const compareNewestFirst = (a: Transaction, b: Transaction) => b.date.localeCompare(a.date) || b.createdAt.localeCompare(a.createdAt);

export function toTransactionViews(
  transactions: Transaction[],
  refs: { wallets: Wallet[]; categories: Category[]; vehicles: Vehicle[] },
): TransactionView[] {
  const wallets = new Map(refs.wallets.map((w) => [w.id, w]));
  const categories = new Map(refs.categories.map((c) => [c.id, c]));
  const vehicles = new Map(refs.vehicles.map((v) => [v.id, v]));
  return [...transactions].sort(compareNewestFirst).map((t) => {
    const category = t.categoryId ? categories.get(t.categoryId) : undefined;
    return {
      ...t,
      categoryName: category?.name ?? null,
      categoryIcon: category?.icon ?? null,
      walletName: wallets.get(t.walletId)?.name ?? "?",
      toWalletName: t.toWalletId ? (wallets.get(t.toWalletId)?.name ?? "?") : null,

  });
```

```ts
import type { Transaction } from "@/modules/transactions/types/transaction";
import type { Wallet } from "../types/wallet";

/** Saldo per dompet (spec §3.3): awal + masuk − keluar + transfer masuk − (transfer keluar + admin). `upTo` inklusif. */
export function walletBalances(wallets: Wallet[], transactions: Transaction[], upTo?: string): Map<string, number> {
  const balances = new Map(wallets.map((w) => [w.id, w.openingBalance]));
  const add = (id: string | null, delta: number) => {
    if (id && balances.has(id)) balances.set(id, balances.get(id)! + delta);
  };
  for (const t of transactions) {
    if (upTo && t.date > upTo) continue;
    if (t.type === "income") add(t.walletId, t.amount);
    else if (t.type === "expense") add(t.walletId, -t.amount);
    else {
      add(t.walletId, -(t.amount + t.adminFee));
      add(t.toWalletId, t.amount);
    }
  }
  return balances;
}
```

```ts
// Pagu (spec §3.3): nilai = tarif × basis × hari; kelebihan pemakaian = ditanggung kantor.
import { sum } from "@/modules/shared/lib/money";
import type { ExpenseEntry } from "@/modules/transactions/types/transaction";
import type { Budget, BudgetUsage, DailyBudgetUsage } from "../types/budget";

export interface BudgetContext {
  pax: number;
  vehicleCount: number;
  days: number;
}

export function budgetContextOf(project: { pax: number; days: number; vehicles: unknown[] }): BudgetContext {
  return { pax: project.pax, vehicleCount: project.vehicles.length, days: project.days };
}

/** Batas per hari; null untuk lump_sum (tidak punya batas harian). */
export function dailyLimit(budget: Pick<Budget, "rate" | "basis">, ctx: BudgetContext): number | null {
  switch (budget.basis) {
    case "per_pax_per_day":
      return budget.rate * ctx.pax;
    case "per_vehicle_per_day":
      return budget.rate * ctx.vehicleCount;
    case "per_day":
      return budget.rate;
    case "lump_sum":
      return null;
  }
}

export function budgetLimit(budget: Pick<Budget, "rate" | "basis">, ctx: BudgetContext): number {
  const daily = dailyLimit(budget, ctx);
  return daily === null ? budget.rate : daily * ctx.days;
}

const usedBy = (budget: Budget, entries: ExpenseEntry[]) => {
  const ids = new Set(budget.categoryIds);
  return sum(entries.filter((e) => e.categoryId !== null && ids.has(e.categoryId)).map((e) => e.amount));
};

export function budgetUsage(budget: Budget, ctx: BudgetContext, entries: ExpenseEntry[]): BudgetUsage {
  const limit = budgetLimit(budget, ctx);
  const used = usedBy(budget, entries);
  return {
    budgetId: budget.id,
    name: budget.name,
    limit,
    used,
    remaining: Math.max(limit - used, 0),
    overBy: Math.max(used - limit, 0),
    ratio: limit > 0 ? used / limit : 0,
  };
}

export function dailyBudgetUsage(budget: Budget, ctx: BudgetContext, entries: ExpenseEntry[], date: string): DailyBudgetUsage | null {
  const limit = dailyLimit(budget, ctx);
  if (limit === null) return null;
  return { budgetId: budget.id, name: budget.name, limit, used: usedBy(budget, entries.filter((e) => e.date === date)) };
}
```

```ts
// Bulan = "YYYY-MM". Grid kalender dimulai Senin (kebiasaan Indonesia).
import { addDays } from "@/modules/shared/lib/date";

const MONTH = /^(\d{4})-(\d{2})$/;

export function isMonth(value: string): boolean {
  const match = MONTH.exec(value);
  return !!match && Number(match[2]) >= 1 && Number(match[2]) <= 12;
}

export const monthOf = (date: string) => date.slice(0, 7);

export function shiftMonth(month: string, delta: number): string {
  const [year, mon] = month.split("-").map(Number);
  return new Date(Date.UTC(year, mon - 1 + delta, 1)).toISOString().slice(0, 7);
}

export interface DayCell {
  date: string;
  inMonth: boolean;
}

/** Minggu-minggu (Senin–Minggu) yang mencakup seluruh bulan. */
export function monthGrid(month: string): DayCell[][] {
  const first = `${month}-01`;
  const mondayOffset = (new Date(`${first}T00:00:00Z`).getUTCDay() + 6) % 7;
  const nextMonthStart = `${shiftMonth(month, 1)}-01`;
  const weeks: DayCell[][] = [];
  for (let cursor = addDays(first, -mondayOffset); cursor < nextMonthStart; cursor = addDays(cursor, 7)) {
    weeks.push(
      Array.from({ length: 7 }, (_, i) => {
        const date = addDays(cursor, i);
        return { date, inMonth: monthOf(date) === month };
      }),
    );
  }
  return weeks;
}

/** Bulan yang dibuka pertama kali: bulan hari ini bila masih dalam periode proyek, selain itu bulan terdekat dari periode. */
export function defaultCalendarMonth(today: string, startDate: string, endDate: string): string {
  if (today < startDate) return monthOf(startDate);
  if (today > endDate) return monthOf(endDate);
  return monthOf(today);
}

const monthLabel = new Intl.DateTimeFormat("id-ID", { month: "long", year: "numeric", timeZone: "UTC" });

/** "Oktober 2026" */
export function formatMonthLabel(month: string): string {
  return monthLabel.format(new Date(`${month}-01T00:00:00Z`));
}
```

```ts
import type { TransactionType } from "@/modules/shared/types/enums";
import type { TransactionView } from "../types/transaction";

/** "" = semua. */
export interface TransactionFilters {
  type: TransactionType | "all";
  categoryId: string;
  walletId: string;
  vehicleId: string;
  q: string;
}

export const EMPTY_FILTERS: TransactionFilters = { type: "all", categoryId: "", walletId: "", vehicleId: "", q: "" };

export function filterTransactions(views: TransactionView[], filters: TransactionFilters): TransactionView[] {
  const q = filters.q.trim().toLowerCase();
  return views.filter(
    (v) =>
      (filters.type === "all" || v.type === filters.type) &&
      (!filters.categoryId || v.categoryId === filters.categoryId) &&
      (!filters.walletId || v.walletId === filters.walletId || v.toWalletId === filters.walletId) &&
      (!filters.vehicleId || v.vehicleId === filters.vehicleId) &&
      (!q || [v.merchant, v.note, v.categoryName].some((text) => text?.toLowerCase().includes(q))),
  );
}

export interface DayGroup {
  date: string;
  items: TransactionView[];
  expenseTotal: number;
}

/** `views` sudah urut terbaru dulu (toTransactionViews). Subtotal = pengeluaran + biaya admin transfer. */
export function groupByDate(views: TransactionView[]): DayGroup[] {
  const groups: DayGroup[] = [];
  for (const view of views) {
    let group = groups.at(-1);
    if (!group || group.date !== view.date) {
      group = { date: view.date, items: [], expenseTotal: 0 };
      groups.push(group);
    }
    group.items.push(view);
    group.expenseTotal += view.type === "expense" ? view.amount : view.type === "transfer" ? view.adminFee : 0;
  }
  return groups;
}
```

```ts
import "server-only";
import { ValidationError, type FieldErrors } from "@/lib/errors";
import { parseOrThrow } from "@/lib/validation";
import type { Category } from "@/modules/budgets/types/category";
import type { Vehicle } from "@/modules/projects/types/project";
import type { TransactionType } from "@/modules/shared/types/enums";
import type { Wallet } from "@/modules/wallets/types/wallet";
import { transactionInputSchema } from "../types/transaction-input";

export interface TransactionRefs {
  wallets: Wallet[];
  categories: Category[];
  vehicles: Vehicle[];
}

export interface TransactionValues {
  type: TransactionType;
  date: string;
  amount: number;
  categoryId: string | null;
  walletId: string;
  toWalletId: string | null;
  adminFee: number;
  vehicleId: string | null;
  merchant: string | null;
  note: string | null;
}

/** Validasi input + aturan bisnis spec §3.4 terhadap referensi proyek; hasil siap disimpan. */
export function planTransaction(input: unknown, refs: TransactionRefs): TransactionValues {
  const v = parseOrThrow(transactionInputSchema, input);
  const errors: FieldErrors = {};
  if (!refs.wallets.some((w) => w.id === v.walletId)) errors.walletId = ["Dompet tidak ditemukan di proyek ini."];

  if (v.type === "transfer") {
    if (!refs.wallets.some((w) => w.id === v.toWalletId)) errors.toWalletId = ["Dompet tujuan tidak ditemukan di proyek ini."];
    if (Object.keys(errors).length) throw new ValidationError(errors);
    return { ...v, categoryId: null, vehicleId: null };
  }

  const category = refs.categories.find((c) => c.id === v.categoryId);
  if (!category) errors.categoryId = ["Kategori tidak ditemukan di proyek ini."];
  else if (category.kind !== v.type) errors.categoryId = ["Kategori tidak cocok dengan jenis transaksi."];
  if (v.vehicleId) {
    if (category && !category.isVehicleRelated) errors.vehicleId = ["Kategori ini tidak terkait mobil."];
    else if (!refs.vehicles.some((vehicle) => vehicle.id === v.vehicleId)) errors.vehicleId = ["Mobil tidak ditemukan di proyek ini."];
  }
  if (Object.keys(errors).length) throw new ValidationError(errors);
  return { ...v, toWalletId: null, adminFee: 0 };
}
```

```ts
export interface DayDetail {
  date: string;
  expenseTotal: number;
  byCategory: { categoryId: string | null; name: string; icon: string | null; total: number }[];
  budgets: DailyBudgetUsage[];
  transactions: TransactionView[];
}

export const getDayDetail = cache(async (db: Db, projectId: string, date: string): Promise<DayDetail> => {
  if (!isIsoDate(date)) throw new ValidationError({ date: ["Tanggal tidak valid."] });
  const ledger = await getLedger(db, projectId);
  const entries = expenseEntries(ledger.transactions, ledger.categories).filter((e) => e.date === date);
  const categories = new Map(ledger.categories.map((c) => [c.id, c]));
  const ctx = budgetContextOf(ledger.project);
  return {
    date,
    expenseTotal: sum(entries.map((e) => e.amount)),
    byCategory: [...sumBy(entries, (e) => e.categoryId, (e) => e.amount)]
      .map(([categoryId, total]) => {
        const category = categoryId ? categories.get(categoryId) : undefined;
        return { categoryId, name: category?.name ?? "Biaya admin", icon: category?.icon ?? "landmark", total };
      })
      .sort((a, b) => b.total - a.total),
    budgets: ledger.budgets.map((b) => dailyBudgetUsage(b, ctx, entries, date)).filter((b): b is DailyBudgetUsage => b !== null),
    transactions: toTransactionViews(
      ledger.transactions.filter((t) => t.date === date),
      refsOf(ledger),
    ),
  };
});
```

```tsx
import { safeNextPath } from "@/modules/auth/lib/safe-next";
import { getActiveProject } from "@/modules/projects/queries/projects";
import { isIsoDate, todayWib } from "@/modules/shared/lib/date";
import { TransactionForm } from "../components/transaction-form";
import { getTransactionReference, getTransactionView } from "../queries/views";
import type { TransactionView } from "../types/transaction";

type SearchParams = Promise<Record<string, string | string[] | undefined>>;

const single = (value: string | string[] | undefined) => (Array.isArray(value) ? value[0] : value);

export async function TransactionFormSection({ searchParams, transactionId }: { searchParams: SearchParams; transactionId?: Promise<string> }) {
  await connection();
  const db = getDb();
  const project = await getActiveProject(db);
  if (!project) redirect("/proyek/baru");

  const params = await searchParams;
  const requestedDate = single(params.tanggal);
  const returnTo = safeNextPath(single(params.dari) ?? (transactionId ? "/transaksi" : "/"));
  const reference = await getTransactionReference(db, project.id);

  let initial: TransactionView | undefined;
  if (transactionId) {
    try {
      initial = await getTransactionView(db, project.id, await transactionId);
    } catch (error) {
      if (error instanceof NotFoundError) notFound();
      throw error;
    }
  }

  return (
    <TransactionForm
      reference={reference}
      initial={initial}
      returnTo={returnTo}
      defaultDate={requestedDate && isIsoDate(requestedDate) ? requestedDate : todayWib()}
    />
  );
}
```

```tsx
          {budget.used > budget.limit && <p className="text-destructive">Lebih {formatRupiah(budget.used - budget.limit)} → ditanggung kantor</p>}
        </div>
      ))}

      {data.byCategory.length > 0 && (
        <ul className="flex flex-col gap-1 text-sm">
          {data.byCategory.map((row) => (
            <li key={row.categoryId ?? "admin"} className="flex items-center gap-2">
              <CategoryIcon name={row.icon} className="size-4" />
              <span className="flex-1">{row.name}</span>
              <span className="tabular-nums">{formatRupiah(row.total)}</span>
            </li>
          ))}
        </ul>
      )}

      {data.transactions.length ? (
        <ul className="divide-y border-t-2">
          {data.transactions.map((view) => (
            <li key={view.id}>
              <TransactionRow view={view} href={`/transaksi/${view.id}?dari=/kalender`} />
            </li>
          ))}
        </ul>
      ) : (
        <p className="text-sm text-muted-foreground">Belum ada transaksi di tanggal ini.</p>
      )}

      <Link href={addHref} className={cn(buttonVariants(), "w-full")}>
        + Catat di tanggal ini
      </Link>
    </div>
  );
}
```

```tsx
import { dehydrate, HydrationBoundary, QueryClient } from "@tanstack/react-query";
import { redirect } from "next/navigation";
import { connection } from "next/server";
import { getDb } from "@/db/client";
import { getActiveProject } from "@/modules/projects/queries/projects";
import { todayWib } from "@/modules/shared/lib/date";
import { ledgerKeys } from "@/modules/transactions/hooks/keys";
import { CalendarMonth } from "../components/calendar-month";
import { defaultCalendarMonth } from "../lib/month";
import { getCalendarMonth } from "../queries/calendar";

export async function CalendarSection() {
  await connection();
  const db = getDb();
  const project = await getActiveProject(db);
  if (!project) redirect("/proyek/baru");
  const today = todayWib();
  const month = defaultCalendarMonth(today, project.startDate, project.endDate);
  const queryClient = new QueryClient();
  queryClient.setQueryData(ledgerKeys.calendar(project.id, month), await getCalendarMonth(db, project.id, month));
  return (
    <HydrationBoundary state={dehydrate(queryClient)}>
      <CalendarMonth projectId={project.id} initialMonth={month} today={today} startDate={project.startDate} endDate={project.endDate} />
    </HydrationBoundary>
  );
}
```
