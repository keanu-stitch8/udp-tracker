```ts
/** Semua pengeluaran: nominal expense + biaya admin expense/transfer (kategori isAdminFee; bukan baris transaksi kedua). */
export function expenseEntries(transactions: Transaction[], categories: Category[]): ExpenseEntry[] {
  const adminCategoryId = categories.find((c) => c.isAdminFee)?.id ?? null;
  const entries: ExpenseEntry[] = [];
  for (const t of transactions) {
    if (t.type === "expense") entries.push({ transactionId: t.id, date: t.date, categoryId: t.categoryId, walletId: t.walletId, amount: t.amount });
    if (t.type !== "income" && t.adminFee > 0) {
      entries.push({ transactionId: t.id, date: t.date, categoryId: adminCategoryId, walletId: t.walletId, amount: t.adminFee });
    }
  }
  return entries;
}
```

```ts
/**
 * Validasi input + aturan bisnis spec §3.4 terhadap referensi proyek; hasil siap disimpan.
 * `existing` = transaksi yang sedang diubah: pengeluaran lama berkategori Biaya Admin tetap boleh disimpan.
 */
export function planTransaction(input: unknown, refs: TransactionRefs, existing?: { categoryId: string | null }): TransactionValues {
  const { attachments, ...v } = parseOrThrow(transactionInputSchema, input);
  void attachments; // diproses terpisah oleh planAttachments, bukan kolom transaksi
  const errors: FieldErrors = {};
  if (!refs.wallets.some((w) => w.id === v.walletId)) errors.walletId = ["Dompet tidak ditemukan di proyek ini."];

  if (v.type === "transfer") {
    if (!refs.wallets.some((w) => w.id === v.toWalletId)) errors.toWalletId = ["Dompet tujuan tidak ditemukan di proyek ini."];
    if (Object.keys(errors).length) throw new ValidationError(errors);
    return { ...v, categoryId: null, vehicleId: null };
  }

  const category = refs.categories.find((c) => c.id === v.categoryId);
  if (!category) errors.categoryId = ["Kategori tidak ditemukan di proyek ini."];
  else if (category.kind !== v.type) errors.categoryId = ["Kategori tidak cocok dengan jenis transaksi."];
  else if (category.isAdminFee && existing?.categoryId !== category.id) errors.categoryId = ["Isi di kolom Biaya admin pada transaksi yang terkena biaya."];
  if (v.type === "income" && v.adminFee > 0) errors.adminFee = ["Pemasukan tidak punya biaya admin."];
  if (v.vehicleId) {
    if (category && !category.isVehicleRelated) errors.vehicleId = ["Kategori ini tidak terkait mobil."];
    else if (!refs.vehicles.some((vehicle) => vehicle.id === v.vehicleId)) errors.vehicleId = ["Mobil tidak ditemukan di proyek ini."];
  }
  if (Object.keys(errors).length) throw new ValidationError(errors);
  return { ...v, toWalletId: null };
}
```

```ts
/** Transaksi yang masuk kategori Biaya Admin: ada biaya admin, atau pengeluaran lama berkategori itu. */
export function isAdminFeeTransaction(t: Transaction, adminCategoryId: string | null): boolean {
  return t.adminFee > 0 || (adminCategoryId !== null && t.categoryId === adminCategoryId);
}

/** Jumlah transaksi per kategori (daftar Kategori). */
export function categoryUseCounts(transactions: Transaction[], categories: Category[]): Map<string, number> {
  const adminCategoryId = categories.find((c) => c.isAdminFee)?.id ?? null;
  const counts = sumBy(
    transactions.filter((t) => t.categoryId && t.categoryId !== adminCategoryId),
    (t) => t.categoryId!,
    () => 1,
  );
  if (adminCategoryId) counts.set(adminCategoryId, transactions.filter((t) => isAdminFeeTransaction(t, adminCategoryId)).length);
  return counts;
}
```

```ts
/** `adminCategoryId`: filter kategori Biaya Admin mencakup biaya admin pengeluaran/transfer. */
export function filterTransactions(views: TransactionView[], filters: TransactionFilters, adminCategoryId: string | null = null): TransactionView[] {
  const q = filters.q.trim().toLowerCase();
  const matchesCategory = (v: TransactionView) =>
    filters.categoryId === adminCategoryId ? isAdminFeeTransaction(v, adminCategoryId) : v.categoryId === filters.categoryId;
  return views.filter(
    (v) =>
      (filters.type === "all" || v.type === filters.type) &&
      (!filters.categoryId || matchesCategory(v)) &&
      (!filters.walletId || v.walletId === filters.walletId || v.toWalletId === filters.walletId) &&
      (!filters.vehicleId || v.vehicleId === filters.vehicleId) &&
      (!q || [v.merchant, v.note, v.categoryName].some((text) => text?.toLowerCase().includes(q))),
  );
}
```
