> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Halaman **Lainnya → Dompet** untuk menambah/mengubah/mengarsipkan dompet, dan halaman detail dompet untuk **cek saldo** (bandingkan saldo asli dengan saldo aplikasi s/d tanggal) lalu **catat selisih** sebagai transaksi penyesuaian (spec §3.3). Saldo dompet di beranda bisa di-tap menuju detail dompet.

**Architecture:** Sama dengan Rencana 2a/2b-1. Tulis: route → command → service (zod + aturan) → repo. **Tech Stack:** sama dengan Rencana 2a.

**Di luar rencana ini (Rencana 2b-3):** kategori, mobil, pengaturan proyek (nama, lokasi, tanggal, pax, Selesai/Buka lagi), **hapus proyek (wajib mengisi password aplikasi/admin)**. Hapus cek saldo tidak dibuat (riwayat hanya informatif).

## Aturan umum

Sama dengan Rencana 1, 2a, 2b-1. Tambahan:

- **Aplikasi generik:** tidak ada angka/aturan khusus satu proyek. Default kategori penyesuaian berdasarkan nama kategori seed umum ("Tol", "Lain-lain", "Lainnya") dengan fallback, dan selalu bisa diganti pengguna.
- Arsip dompet bisa dibalik (bukan hapus), jadi tidak perlu konfirmasi dua langkah. Dompet tidak bisa dihapus karena transaksi mereferensikannya.
- Tombol "Catat selisih" hanya muncul di cek saldo **terbaru** yang selisihnya belum dicatat, untuk mencegah selisih lama dicatat dua kali.

## Keputusan

| Hal | Keputusan |
| --- | --- |
| Saldo aplikasi saat cek | `walletBalances([wallet], transaksi aktif, s/d tanggal cek)`, disimpan sebagai snapshot `computedBalance` |
| Selisih | `actualBalance − computedBalance` |
| Selisih < 0 | expense \|selisih\|, kategori pilihan (default Tol untuk Flazz, Lain-lain untuk lainnya), mobil = mobil dompet bila kategori terkait mobil, catatan "Penyesuaian cek saldo" |
| Selisih > 0 | income, kategori pilihan (default Lainnya, bukan kategori UDP kantor) |
| Selisih 0 | 409 "Saldo sudah cocok" |
| Sudah dicatat (transaksi aktif dengan `balanceCheckId` ini) | 409. Jika transaksi penyesuaian dihapus, selisih boleh dicatat lagi |
| Pratinjau di form | Klien menghitung saldo s/d tanggal dari `balanceHistory` (saldo akhir per tanggal) tanpa request tambahan |

```ts
export interface BalancePoint {
  date: string;
  balance: number;
}

/** Saldo akhir hari satu dompet, hanya tanggal yang ada mutasinya, urut naik. */
export function dailyBalances(wallet: Wallet, transactions: Transaction[]): BalancePoint[] {
  const deltas = new Map<string, number>();
  const add = (date: string, delta: number) => deltas.set(date, (deltas.get(date) ?? 0) + delta);
  for (const t of transactions) {
    if (t.type === "income" && t.walletId === wallet.id) add(t.date, t.amount);
    else if (t.type === "expense" && t.walletId === wallet.id) add(t.date, -t.amount);
    else if (t.type === "transfer") {
      if (t.walletId === wallet.id) add(t.date, -(t.amount + t.adminFee));
      if (t.toWalletId === wallet.id) add(t.date, t.amount);
    }
  }
  let balance = wallet.openingBalance;
  return [...deltas.keys()].sort().map((date) => {
    balance += deltas.get(date)!;
    return { date, balance };
  });
}

/** Saldo s/d tanggal (inklusif) dari hasil dailyBalances. */
export function balanceAt(points: BalancePoint[], openingBalance: number, date: string): number {
  let balance = openingBalance;
  for (const point of points) {
    if (point.date > date) break;
    balance = point.balance;
  }
  return balance;
}
```

```ts
// Cek saldo (spec §3.3): selisih = saldo asli − saldo aplikasi; "Catat selisih" membuat transaksi penyesuaian.
import type { Category } from "@/modules/budgets/types/category";
import { formatRupiah } from "@/modules/shared/lib/money";
import type { TransactionInput } from "@/modules/transactions/types/transaction-input";
import type { Wallet } from "../types/wallet";

export const ADJUSTMENT_NOTE = "Penyesuaian cek saldo";

/** Kategori bawaan untuk selisih: kurang → pengeluaran (Tol untuk Flazz, selain itu Lain-lain), lebih → pemasukan bukan UDP kantor. */
export function defaultAdjustmentCategoryId(difference: number, wallet: Pick<Wallet, "kind">, categories: Category[]): string | null {
  const active = categories.filter((c) => !c.archived);
  if (difference > 0) {
    const income = active.filter((c) => c.kind === "income" && !c.isOfficeFunding);
    return (income.find((c) => c.name === "Lainnya") ?? income[0])?.id ?? null;
  }
  const expense = active.filter((c) => c.kind === "expense" && !c.isAdminFee);
  const preferred = wallet.kind === "flazz" ? "Tol" : "Lain-lain";
  return (expense.find((c) => c.name === preferred) ?? expense.find((c) => c.name === "Lain-lain") ?? expense[0])?.id ?? null;
}

export function adjustmentInput(
  check: { date: string; difference: number },
  wallet: Pick<Wallet, "id" | "vehicleId">,
  category: Pick<Category, "id" | "isVehicleRelated">,
): TransactionInput {
  return {
    type: check.difference < 0 ? "expense" : "income",
    date: check.date,
    amount: Math.abs(check.difference),
    categoryId: category.id,
    walletId: wallet.id,
    vehicleId: category.isVehicleRelated ? wallet.vehicleId : null,
    note: ADJUSTMENT_NOTE,
  };
}

export function describeDifference(difference: number): string {
  if (difference === 0) return "Cocok";
  return `${difference < 0 ? "Kurang" : "Lebih"} ${formatRupiah(Math.abs(difference))}`;
}
```

```ts
export function planWallet(input: unknown, refs: { vehicles: Vehicle[]; otherWalletNames: string[] }): WalletValues {
  const v = parseOrThrow(walletInputSchema, input);
  const errors: FieldErrors = {};
  if (refs.otherWalletNames.some((n) => n.toLowerCase() === v.name.toLowerCase())) errors.name = ["Nama dompet sudah dipakai."];

  }
  if (Object.keys(errors).length) throw new ValidationError(errors);
  return v;
}
```

```ts
/** "Catat selisih" (spec §3.3): satu transaksi penyesuaian aktif per cek saldo. */
export async function adjustBalanceCheck(db: Db, projectId: string, checkId: string, input: unknown): Promise<Transaction> {
  await editableProject(db, projectId);
  const check = await findBalanceCheckOrThrow(db, projectId, checkId);
  const difference = check.actualBalance - check.computedBalance;
  if (difference === 0) throw new ConflictError("Saldo sudah cocok, tidak ada selisih untuk dicatat.");

  const [walletRows, categoryRows, vehicleRows, transactionRows] = await Promise.all([
    selectWallets(db, projectId),
    selectCategories(db, projectId),
    selectVehicles(db, projectId),
    selectTransactions(db, projectId),
  ]);
  if (transactionRows.some((t) => t.balanceCheckId === check.id)) throw new ConflictError("Selisih cek saldo ini sudah dicatat.");

  const wallets = walletRows.map(toWallet);
  const categories = categoryRows.map(toCategory);
  const wallet = wallets.find((w) => w.id === check.walletId)!;
  const { categoryId } = parseOrThrow(adjustInputSchema, input);
  const category = categories.find((c) => c.id === (categoryId ?? defaultAdjustmentCategoryId(difference, wallet, categories)));
  if (!category) throw new ValidationError({ categoryId: ["Pilih kategori."] });

  const values = planTransaction(adjustmentInput({ date: check.date, difference }, wallet, category), {
    wallets,
    categories,
    vehicles: vehicleRows.map(toVehicle),
  });
  return toTransaction(await insertTransaction(db, { ...values, projectId, balanceCheckId: check.id }));
}
```

```tsx
"use client";

import { useState } from "react";
import { toast } from "sonner";
import { NATIVE_SELECT_CLASS } from "@/components/app/form-classes";
import { MoneyInput } from "@/components/app/money-input";
import { Button } from "@/components/ui/button";
import { Field, FieldDescription, FieldError, FieldGroup, FieldLabel } from "@/components/ui/field";
import { Input } from "@/components/ui/input";
import { NativeSelect, NativeSelectOption } from "@/components/ui/native-select";
import { ApiError } from "@/lib/api-client";
import type { Vehicle } from "@/modules/projects/types/project";
import { WALLET_KINDS, type WalletKind } from "@/modules/shared/types/enums";
import { useSaveWallet } from "../hooks/use-wallets";
import { WALLET_KIND_LABELS } from "../lib/kinds";
import type { Wallet } from "../types/wallet";

interface WalletFormProps {
  projectId: string;
  vehicles: Vehicle[];
  initial?: Wallet;
  onDone: () => void;
}

export function WalletForm({ projectId, vehicles, initial, onDone }: WalletFormProps) {
  const save = useSaveWallet(projectId, initial?.id);
  const [kind, setKind] = useState<WalletKind>(initial?.kind ?? "cash");
  const [name, setName] = useState(initial?.name ?? "");
  const [holder, setHolder] = useState(initial?.holder ?? "");
  const [vehicleId, setVehicleId] = useState(initial?.vehicleId ?? "");
  const [openingBalance, setOpeningBalance] = useState<number | null>(initial?.openingBalance ?? 0);

  const fieldErrors = save.error instanceof ApiError ? save.error.fieldErrors : {};
  const errorOf = (field: string) => fieldErrors[field]?.[0];
  const showVehicle = kind === "flazz" && vehicles.length > 0;

  return (
    <form
      className="flex flex-col gap-4"
      onSubmit={(event) => {
        event.preventDefault();
        save.mutate(
          { kind, name, holder, vehicleId: showVehicle && vehicleId ? vehicleId : null, openingBalance: openingBalance ?? 0, archived: initial?.archived ?? false },
          {
            onSuccess: () => {
              toast.success(initial ? "Dompet diperbarui." : "Dompet ditambahkan.");
              onDone();
            },
          },
        );
      }}
    >
      <FieldGroup>
        <div className="grid grid-cols-2 gap-3">
          <Field data-invalid={errorOf("kind") ? true : undefined}>
            <FieldLabel htmlFor="wallet-kind">Jenis</FieldLabel>
            <NativeSelect id="wallet-kind" className={NATIVE_SELECT_CLASS} value={kind} onChange={(e) => setKind(e.target.value as WalletKind)}>
              {WALLET_KINDS.map((k) => (
                <NativeSelectOption key={k} value={k}>
                  {WALLET_KIND_LABELS[k]}
                </NativeSelectOption>
              ))}
            </NativeSelect>
            {errorOf("kind") && <FieldError>{errorOf("kind")}</FieldError>}
          </Field>
          <Field data-invalid={errorOf("name") ? true : undefined}>
            <FieldLabel htmlFor="wallet-name">Nama</FieldLabel>
            <Input id="wallet-name" className="h-10" value={name} onChange={(e) => setName(e.target.value)} placeholder="Cash Tim" />
            {errorOf("name") && <FieldError>{errorOf("name")}</FieldError>}
          </Field>
        </div>
        {showVehicle && (
          <Field data-invalid={errorOf("vehicleId") ? true : undefined}>
            <FieldLabel htmlFor="wallet-vehicle">Mobil</FieldLabel>
            <NativeSelect id="wallet-vehicle" className={NATIVE_SELECT_CLASS} value={vehicleId} onChange={(e) => setVehicleId(e.target.value)}>
              <NativeSelectOption value="">Tanpa mobil</NativeSelectOption>
              {vehicles.map((v) => (
                <NativeSelectOption key={v.id} value={v.id}>
                  {v.name}
                </NativeSelectOption>
              ))}
            </NativeSelect>
            {errorOf("vehicleId") && <FieldError>{errorOf("vehicleId")}</FieldError>}
          </Field>
        )}
        <Field data-invalid={errorOf("holder") ? true : undefined}>
          <FieldLabel htmlFor="wallet-holder">Pemegang</FieldLabel>
          <Input id="wallet-holder" className="h-10" value={holder} onChange={(e) => setHolder(e.target.value)} placeholder="Opsional" />
          {errorOf("holder") && <FieldError>{errorOf("holder")}</FieldError>}
        </Field>
        <Field data-invalid={errorOf("openingBalance") ? true : undefined}>
          <FieldLabel htmlFor="wallet-opening">Saldo awal</FieldLabel>
          <MoneyInput id="wallet-opening" className="h-10" value={openingBalance} onChange={setOpeningBalance} />
          <FieldDescription>Saldo saat dompet mulai dicatat di aplikasi.</FieldDescription>
          {errorOf("openingBalance") && <FieldError>{errorOf("openingBalance")}</FieldError>}
        </Field>
      </FieldGroup>
      {save.error && !Object.keys(fieldErrors).length && (
        <p role="alert" className="text-sm text-destructive">
          {save.error.message}
        </p>
      )}
      <div className="flex gap-2">
        <Button type="submit" className="flex-1" disabled={save.isPending}>
          {save.isPending ? "Menyimpan…" : "Simpan"}
        </Button>
        <Button type="button" variant="outline" onClick={onDone}>
          Batal
        </Button>
      </div>
    </form>
  );
}
```

```tsx
import { ChevronRight } from "lucide-react";
import Link from "next/link";
import { useState } from "react";
import { Alert, AlertDescription, AlertTitle } from "@/components/ui/alert";
import { Button } from "@/components/ui/button";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { cn } from "@/lib/utils";

import { useWalletList } from "../hooks/use-wallets";
import { WALLET_KIND_LABELS } from "../lib/kinds";
import type { WalletListItem } from "../queries/wallets";
import { WalletForm } from "./wallet-form";

export function WalletList({ projectId }: { projectId: string }) {
  const { data } = useWalletList(projectId);
  const [adding, setAdding] = useState(false);
  const [showArchived, setShowArchived] = useState(false);
  if (!data) return null;
  const active = data.wallets.filter((w) => !w.archived);
  const archived = data.wallets.filter((w) => w.archived);

  return (
    <div className="flex flex-col gap-4">
      {data.locked && (
        <Alert>
          <AlertTitle>Proyek sudah selesai</AlertTitle>
          <AlertDescription>Dompet dikunci. Buka kembali proyek untuk mengubah.</AlertDescription>
        </Alert>
      )}
      <p className="text-sm text-muted-foreground">Tap dompet untuk cek saldo dan mencocokkan dengan saldo asli.</p>

      {adding ? (
        <Card>
          <CardHeader>
            <CardTitle>Dompet baru</CardTitle>
          </CardHeader>
          <CardContent>
            <WalletForm projectId={projectId} vehicles={data.vehicles} onDone={() => setAdding(false)} />
          </CardContent>
        </Card>
      ) : (
        <Button variant="outline" disabled={data.locked} onClick={() => setAdding(true)}>
          + Tambah dompet
        </Button>
      )}

      {active.length ? <WalletLinks wallets={active} /> : <p className="text-sm text-muted-foreground">Belum ada dompet aktif.</p>}

      {archived.length > 0 && (
        <>
          <Button variant="ghost" className="self-start" onClick={() => setShowArchived((v) => !v)}>
            {showArchived ? "Sembunyikan" : "Tampilkan"} dompet diarsipkan ({archived.length})
          </Button>
          {showArchived && <WalletLinks wallets={archived} />}
        </>
      )}
    </div>
  );
}

function WalletLinks({ wallets }: { wallets: WalletListItem[] }) {
  return (
    <ul className="flex flex-col divide-y-2 border-2 bg-card shadow-md">
      {wallets.map((wallet) => (
        <li key={wallet.id}>
          <Link href={`/lainnya/dompet/${wallet.id}`} className="flex min-h-14 items-center gap-3 px-4 py-2">
            <div className="min-w-0 flex-1">
              <p className="truncate font-medium">{wallet.name}</p>
              <p className="truncate text-xs text-muted-foreground">{[WALLET_KIND_LABELS[wallet.kind], wallet.vehicleName, wallet.holder].filter(Boolean).join(" · ")}</p>
            </div>
            <span className={cn("font-head tabular-nums", wallet.balance < 0 && "text-destructive")}>{formatRupiah(wallet.balance)}</span>
            <ChevronRight className="size-4 shrink-0" />
          </Link>
        </li>
      ))}
    </ul>
  );
}
```

```tsx
"use client";

import Link from "next/link";
import { useState } from "react";
import { toast } from "sonner";
import { CategoryIcon } from "@/components/app/category-icon";
import { Button } from "@/components/ui/button";
import { Toggle } from "@/components/ui/toggle";
import { cn } from "@/lib/utils";
import { formatDateShort } from "@/modules/shared/lib/date";
import { formatRupiah } from "@/modules/shared/lib/money";
import { useAdjustBalanceCheck } from "../hooks/use-wallets";
import { defaultAdjustmentCategoryId, describeDifference } from "../lib/balance-check";
import type { WalletDetail } from "../queries/wallets";
import type { BalanceCheck } from "../types/balance-check";

/** Satu baris riwayat cek saldo. "Catat selisih" hanya untuk cek terbaru (`canAdjust`) agar selisih lama tidak tercatat dua kali. */
export function BalanceCheckItem({ check, detail, canAdjust }: { check: BalanceCheck; detail: WalletDetail; canAdjust: boolean }) {
  const adjust = useAdjustBalanceCheck(detail.projectId);
  const [open, setOpen] = useState(false);
  const [categoryId, setCategoryId] = useState(() => defaultAdjustmentCategoryId(check.difference, detail.wallet, detail.categories) ?? "");
  const kind = check.difference < 0 ? "expense" : "income";
  const options = detail.categories.filter((c) => c.kind === kind && !c.isAdminFee && !c.isOfficeFunding);

  return (
    <li className="flex flex-col gap-2 py-3 text-sm">
      <div className="flex items-center justify-between gap-2">
        <span className="font-medium">{formatDateShort(check.date)}</span>
        <span className={cn("font-head tabular-nums", check.difference < 0 && "text-destructive", check.difference > 0 && "text-green-700")}>
          {describeDifference(check.difference)}
        </span>
      </div>
      <p className="text-xs text-muted-foreground">
        Asli {formatRupiah(check.actualBalance)} · aplikasi {formatRupiah(check.computedBalance)}
        {check.note ? ` · ${check.note}` : ""}
      </p>
      {check.difference !== 0 &&
        (check.adjustmentId ? (
          <Link href={`/transaksi/${check.adjustmentId}?dari=/lainnya/dompet/${detail.wallet.id}`} className="self-start text-xs underline">
            Selisih sudah dicatat. Lihat transaksi
          </Link>
        ) : !canAdjust ? null : open ? (
          <div className="flex flex-col gap-2 border-2 p-3">
            <p>
              Catat {formatRupiah(Math.abs(check.difference))} sebagai {kind === "expense" ? "pengeluaran" : "pemasukan"}:
            </p>
            <div className="flex flex-wrap gap-2">
              {options.map((c) => (
                <Toggle key={c.id} pressed={categoryId === c.id} onPressedChange={() => setCategoryId(c.id)} className="h-9 px-2.5">
                  <CategoryIcon name={c.icon} className="size-4" />
                  {c.name}
                </Toggle>
              ))}
            </div>
            <div className="flex gap-2">
              <Button
                size="sm"
                disabled={adjust.isPending || !categoryId || detail.locked}
                onClick={() =>
                  adjust.mutate(
                    { checkId: check.id, categoryId },
                    {
                      onSuccess: () => {
                        toast.success("Selisih dicatat sebagai transaksi.");
                        setOpen(false);
                      },
                      onError: (error) => toast.error(error.message),
                    },
                  )
                }
              >
                {adjust.isPending ? "Mencatat…" : "Catat"}
              </Button>
              <Button size="sm" variant="outline" onClick={() => setOpen(false)}>
                Batal
              </Button>
            </div>
          </div>
        ) : (
          <Button size="sm" variant="outline" className="self-start" disabled={detail.locked} onClick={() => setOpen(true)}>
            Catat selisih
          </Button>
        ))}
    </li>
  );
}
```

```tsx
import { ApiError } from "@/lib/api-client";
import { cn } from "@/lib/utils";
import { formatRupiah } from "@/modules/shared/lib/money";
import { useCreateBalanceCheck, useSaveWallet, useWalletDetail } from "../hooks/use-wallets";

export function WalletDetailPanel({ projectId, walletId, today }: { projectId: string; walletId: string; today: string }) {
  const { data } = useWalletDetail(projectId, walletId);
  const save = useSaveWallet(projectId, walletId);
  const [editing, setEditing] = useState(false);
  if (!data) return null;
  const { wallet } = data;

  function toggleArchived() {
    const archived = !wallet.archived;
    save.mutate(
      { kind: wallet.kind, name: wallet.name, holder: wallet.holder, vehicleId: wallet.vehicleId, openingBalance: wallet.openingBalance, archived },
      {
        onSuccess: () => toast(archived ? "Dompet diarsipkan. Tidak muncul lagi di form transaksi." : "Dompet diaktifkan lagi."),

        <Alert>
          <AlertTitle>Proyek sudah selesai</AlertTitle>
          <AlertDescription>Dompet dan cek saldo dikunci. Buka kembali proyek untuk mengubah.</AlertDescription>
        </Alert>
      )}

      {editing ? (
        <Card>
          <CardHeader>
            <CardTitle>Ubah {wallet.name}</CardTitle>
          </CardHeader>
          <CardContent>
            <WalletForm projectId={projectId} vehicles={data.vehicles} initial={wallet} onDone={() => setEditing(false)} />
          </CardContent>
        </Card>
      ) : (
        <Card>
          <CardHeader>
            <CardTitle>{wallet.name}</CardTitle>
            <CardDescription>{[WALLET_KIND_LABELS[wallet.kind], wallet.vehicleName, wallet.holder].filter(Boolean).join(" · ")}</CardDescription>
            <CardAction>
              <Button size="sm" variant="outline" disabled={data.locked} onClick={() => setEditing(true)}>
                Ubah
              </Button>

          <CardContent className="flex flex-col gap-3">
            <div>
              <p className="text-xs text-muted-foreground">Saldo di aplikasi</p>
              <p className={cn("font-head text-3xl tabular-nums", wallet.balance < 0 && "text-destructive")}>{formatRupiah(wallet.balance)}</p>
              <p className="text-xs text-muted-foreground">Saldo awal {formatRupiah(wallet.openingBalance)}</p>
            </div>
            <div className="flex items-center gap-2">
              {wallet.archived && <Badge variant="outline">Diarsipkan</Badge>}
              <Button size="sm" variant="ghost" disabled={data.locked || save.isPending} onClick={toggleArchived}>
                {wallet.archived ? "Aktifkan lagi" : "Arsipkan dompet"}
              </Button>
            </div>
          </CardContent>
        </Card>
      )}

      <BalanceCheckForm detail={data} today={today} />

      <Card>
        <CardHeader>
          <CardTitle className="text-base">Riwayat cek saldo</CardTitle>
        </CardHeader>
        <CardContent>

function BalanceCheckForm({ detail, today }: { detail: WalletDetail; today: string }) {
  const create = useCreateBalanceCheck(detail.projectId, detail.wallet.id);
  const [date, setDate] = useState(today);
  const [actual, setActual] = useState<number | null>(null);
  const [note, setNote] = useState("");
  const computed = balanceAt(detail.balanceHistory, detail.wallet.openingBalance, date);
  const fieldErrors = create.error instanceof ApiError ? create.error.fieldErrors : {};
  const errorOf = (field: string) => fieldErrors[field]?.[0];

  return (
    <Card>
      <CardHeader>
        <CardTitle className="text-base">Cek saldo</CardTitle>
        <CardDescription>Masukkan saldo asli (cek kartu Flazz, hitung uang cash, atau lihat saldo rekening) untuk dicocokkan.</CardDescription>
      </CardHeader>
      <CardContent>
        <form
          className="flex flex-col gap-4"
          onSubmit={(event) => {
            event.preventDefault();
            if (actual === null) return;
            create.mutate(
              { date, actualBalance: actual, note },
              {
                onSuccess: (check) => {
                  toast(check.difference === 0 ? "Saldo cocok." : `Cek saldo tersimpan: ${describeDifference(check.difference).toLowerCase()}.`);
                  setActual(null);
                  setNote("");
                },
              },
            );
          }}
        >
          <FieldGroup>
            <div className="grid grid-cols-2 gap-3">
              <Field data-invalid={errorOf("date") ? true : undefined}>
                <FieldLabel htmlFor="check-date">Tanggal</FieldLabel>
                <Input id="check-date" type="date" className="h-10" value={date} onChange={(e) => setDate(e.target.value)} />
                {errorOf("date") && <FieldError>{errorOf("date")}</FieldError>}
              </Field>
              <Field data-invalid={errorOf("actualBalance") ? true : undefined}>
                <FieldLabel htmlFor="check-actual">Saldo asli</FieldLabel>
                <MoneyInput id="check-actual" className="h-10" value={actual} onChange={setActual} />
                {errorOf("actualBalance") && <FieldError>{errorOf("actualBalance")}</FieldError>}
              </Field>
            </div>
            <Field data-invalid={errorOf("note") ? true : undefined}>
              <FieldLabel htmlFor="check-note">Catatan</FieldLabel>
              <Input id="check-note" className="h-10" value={note} onChange={(e) => setNote(e.target.value)} placeholder="Opsional" />

            <p role="alert" className="text-sm text-destructive">
              {create.error.message}
            </p>

          <Button type="submit" disabled={create.isPending || actual === null || detail.locked}>
            {create.isPending ? "Menyimpan…" : "Simpan cek saldo"}
          </Button>
        </form>
      </CardContent>
    </Card>
  );
}
```

```tsx
import { dehydrate, HydrationBoundary, QueryClient } from "@tanstack/react-query";
import { notFound, redirect } from "next/navigation";
import { connection } from "next/server";
import { getDb } from "@/db/client";
import { NotFoundError } from "@/lib/errors";
import { getActiveProject } from "@/modules/projects/queries/projects";
import { todayWib } from "@/modules/shared/lib/date";
import { ledgerKeys } from "@/modules/transactions/hooks/keys";
import { WalletDetailPanel } from "../components/wallet-detail";
import { getWalletDetail, type WalletDetail } from "../queries/wallets";

export async function WalletDetailSection({ walletId }: { walletId: Promise<string> }) {
  await connection();
  const db = getDb();
  const project = await getActiveProject(db);
  if (!project) redirect("/proyek/baru");
  const id = await walletId;
  let detail: WalletDetail;
  try {
    detail = await getWalletDetail(db, project.id, id);
  } catch (error) {
    if (error instanceof NotFoundError) notFound();
    throw error;
  }
  const queryClient = new QueryClient();
  queryClient.setQueryData(ledgerKeys.wallet(project.id, id), detail);
  return (
    <HydrationBoundary state={dehydrate(queryClient)}>
      <WalletDetailPanel projectId={project.id} walletId={id} today={todayWib()} />
    </HydrationBoundary>
  );
}
```
