// Progres proyek untuk beranda. Tanggal ISO "YYYY-MM-DD" (WIB), dibandingkan sebagai string.
import { daysInclusive } from "@/modules/shared/lib/date";

export interface ProjectProgress {
  phase: "before" | "during" | "after";
  /** Hari ke-N (1-based); 0 sebelum mulai, totalDays setelah selesai. */
  day: number;
  totalDays: number;
  /** Hari menuju mulai; 0 kecuali phase "before". */
  daysUntil: number;
}

export function projectProgress(startDate: string, endDate: string, today: string): ProjectProgress {
  const totalDays = daysInclusive(startDate, endDate);
  if (today < startDate) return { phase: "before", day: 0, totalDays, daysUntil: daysInclusive(today, startDate) - 1 };
  if (today > endDate) return { phase: "after", day: totalDays, totalDays, daysUntil: 0 };
  return { phase: "during", day: daysInclusive(startDate, today), totalDays, daysUntil: 0 };
}

/** Hari yang sudah berjalan, pembagi rata-rata per hari. */
export function elapsedDays(progress: ProjectProgress): number {
  return progress.day;
}

export function progressLabel(progress: ProjectProgress): string {
  if (progress.phase === "before") return `Mulai ${progress.daysUntil} hari lagi`;
  if (progress.phase === "after") return `Selesai · ${progress.totalDays} hari`;
  return `Hari ke-${progress.day} dari ${progress.totalDays}`;
}
```

import type { Category } from "@/modules/budgets/types/category";
import { sum } from "@/modules/shared/lib/money";
import { sumBy } from "@/modules/transactions/lib/ledger";
import type { ExpenseEntry } from "@/modules/transactions/types/transaction";

export interface CategoryShare {
  /** id kategori, "none" (tanpa kategori), atau "other" (gabungan sisa). */
  key: string;
  name: string;
  icon: string | null;
  amount: number;
  /** Persen dari total pengeluaran, dibulatkan. */
  share: number;
}

/** Pengeluaran per kategori, terbesar dulu; lebih dari `limit` → sisanya jadi "Lainnya". Bukan terhadap pagu. */
export function categoryShares(entries: ExpenseEntry[], categories: Category[], limit = 5): CategoryShare[] {
  const total = sum(entries.map((e) => e.amount));
  const byId = new Map(categories.map((c) => [c.id, c]));
  const share = (amount: number) => (total > 0 ? Math.round((amount / total) * 100) : 0);
  const rows: CategoryShare[] = [...sumBy(entries, (e) => e.categoryId, (e) => e.amount)]
    .map(([id, amount]) => {
      const category = id ? byId.get(id) : undefined;
      return { key: id ?? "none", name: category?.name ?? "Tanpa kategori", icon: category?.icon ?? null, amount, share: share(amount) };
    })
    .sort((a, b) => b.amount - a.amount);
  if (rows.length <= limit) return rows;
  const rest = sum(rows.slice(limit).map((r) => r.amount));
  return [...rows.slice(0, limit), { key: "other", name: "Lainnya", icon: "ellipsis", amount: rest, share: share(rest) }];
}
```

import { formatRupiah } from "@/modules/shared/lib/money";
import { describeDifference } from "@/modules/wallets/lib/balance-check";
import { reimburseStatus } from "@/modules/wallets/lib/reimburse";
import type { WalletBalance } from "@/modules/wallets/types/wallet";

export interface AttentionItem {
  kind: "receipt" | "difference" | "negative" | "reimburse";
  label: string;
  href: string;
}

/** Cek saldo terbaru satu dompet; `adjusted` = selisihnya sudah dicatat sebagai transaksi. */
export interface LatestCheck {
  walletId: string;
  difference: number;
  adjusted: boolean;
}

/** `wallets` = dompet aktif saja. Urutan: bon, selisih cek saldo, dompet minus, reimburse. */
export function attentionItems(input: { missingReceipts: number; wallets: WalletBalance[]; latestChecks: LatestCheck[] }): AttentionItem[] {
  const walletHref = (id: string) => `/lainnya/dompet/${id}`;
  const byId = new Map(input.wallets.map((w) => [w.id, w]));
  const items: AttentionItem[] = [];
  if (input.missingReceipts > 0) {
    items.push({ kind: "receipt", label: `${input.missingReceipts} pengeluaran belum ada bon`, href: "/transaksi?bon=belum" });
  }
  for (const check of input.latestChecks) {
    const wallet = byId.get(check.walletId);
    if (!wallet || check.difference === 0 || check.adjusted) continue;
    items.push({ kind: "difference", label: `${wallet.name}: selisih ${describeDifference(check.difference).toLowerCase()} belum dicatat`, href: walletHref(wallet.id) });
  }
  for (const wallet of input.wallets) {
    if (wallet.kind !== "reimburse" && wallet.balance < 0) {
      items.push({ kind: "negative", label: `${wallet.name} minus ${formatRupiah(-wallet.balance)}`, href: walletHref(wallet.id) });
    }
  }
  for (const wallet of input.wallets) {
    if (wallet.kind === "reimburse" && wallet.balance !== 0) {
      items.push({ kind: "reimburse", label: `${wallet.name}: ${reimburseStatus(wallet.balance)}`, href: walletHref(wallet.id) });
    }
  }
  return items;
}
```

export async function selectProjectBalanceChecks(db: Db, projectId: string): Promise<BalanceCheckRow[]> {
  const rows = await db
    .select({ check: balanceChecks })
    .from(balanceChecks)
    .innerJoin(wallets, eq(wallets.id, balanceChecks.walletId))
    .where(eq(wallets.projectId, projectId))

  for (const row of checkRows) {
    if (seen.has(row.walletId)) continue;
    seen.add(row.walletId);
    latestChecks.push({ walletId: row.walletId, difference: row.actualBalance - row.computedBalance, adjusted: adjusted.has(row.id) });
  }

  return {
    totals,
    progress,
    todayExpense: sum(entries.filter((e) => e.date === today).map((e) => e.amount)),
    dailyAverage: elapsed > 0 ? Math.round(totals.totalExpense / elapsed) : null,
    attention: attentionItems({ missingReceipts: views.filter((v) => isMissingReceipt(v, adminCategoryId)).length, wallets, latestChecks }),
    categories: categoryShares(entries, ledger.categories),
    wallets,
    recent: views.slice(0, 5),
  };
});
```
