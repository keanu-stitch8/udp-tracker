// Format & parse rupiah. Nominal selalu integer rupiah (spec §3).
const idNumber = new Intl.NumberFormat("id-ID", { maximumFractionDigits: 0 });

export function formatRupiah(amount: number): string {
  const sign = amount < 0 ? "-" : "";
  return `${sign}Rp ${idNumber.format(Math.abs(amount))}`;
}

const oneDecimal = (n: number) => (Math.round(n * 10) / 10).toString().replace(".", ",").replace(/,0$/, "");

/** Ringkas untuk ruang sempit (sel kalender): 850rb, 1,3jt, 1,5M. */
export function formatRupiahCompact(amount: number): string {
  const sign = amount < 0 ? "-" : "";
  const abs = Math.abs(amount);
  if (abs >= 1_000_000_000) return `${sign}${oneDecimal(abs / 1_000_000_000)}M`;
  if (abs >= 1_000_000) return `${sign}${oneDecimal(abs / 1_000_000)}jt`;
  if (abs >= 1_000) return `${sign}${oneDecimal(abs / 1_000)}rb`;
  return `${sign}${abs}`;
}

/** "Rp 1.350.000" / "75.000,00" / "1,350,000" → integer rupiah; null jika tidak ada angka. */
export function parseRupiah(input: string): number | null {
  const withoutDecimals = input.trim().replace(/,\d{1,2}$/, "");
  const digits = withoutDecimals.replace(/\D/g, "");
  return digits ? Number(digits) : null;
}
```

// Tanggal transaksi = string "YYYY-MM-DD" dalam WIB (spec §3). Aritmetika memakai UTC agar bebas zona waktu mesin.
import { z } from "zod";

export const WIB = "Asia/Jakarta";
const DAY_MS = 86_400_000;
const ISO_DATE = /^(\d{4})-(\d{2})-(\d{2})$/;

const wibDate = new Intl.DateTimeFormat("en-CA", { timeZone: WIB });

/** Tanggal hari ini di WIB, format YYYY-MM-DD. */
export function todayWib(now: Date = new Date()): string {
  return wibDate.format(now);
}

function toUtcMs(iso: string): number {
  const [, y, m, d] = ISO_DATE.exec(iso)!;
  return Date.UTC(Number(y), Number(m) - 1, Number(d));
}

export function isIsoDate(value: string): boolean {
  const match = ISO_DATE.exec(value);
  if (!match) return false;
  const date = new Date(toUtcMs(value));
  return date.getUTCFullYear() === Number(match[1]) && date.getUTCMonth() + 1 === Number(match[2]) && date.getUTCDate() === Number(match[3]);
}

export const isoDateSchema = z.string().refine(isIsoDate, "Tanggal tidak valid.");

/** Jumlah hari dari start s/d end, inklusif (20–25 Okt = 6 hari). */
export function daysInclusive(start: string, end: string): number {
  return Math.round((toUtcMs(end) - toUtcMs(start)) / DAY_MS) + 1;
}

export function addDays(iso: string, days: number): string {
  return new Date(toUtcMs(iso) + days * DAY_MS).toISOString().slice(0, 10);
}

const shortFormat = new Intl.DateTimeFormat("id-ID", { weekday: "short", day: "numeric", month: "short", timeZone: "UTC" });
const longFormat = new Intl.DateTimeFormat("id-ID", { day: "numeric", month: "short", year: "numeric", timeZone: "UTC" });

/** "Sel, 20 Okt" */
export function formatDateShort(iso: string): string {
  return shortFormat.format(new Date(toUtcMs(iso)));
}

/** "20 Okt 2026" */
export function formatDateLong(iso: string): string {
  return longFormat.format(new Date(toUtcMs(iso)));
}
```

export class AppError extends Error {
  readonly status: number;
  constructor(message: string, status: number) {
    super(message);
    this.name = new.target.name;
    this.status = status;
  }
}

export type FieldErrors = Record<string, string[]>;

export class ValidationError extends AppError {
  readonly fieldErrors: FieldErrors;
  constructor(fieldErrors: FieldErrors, message = "Periksa kembali isian yang ditandai.") {
    super(message, 422);
    this.fieldErrors = fieldErrors;
  }
}

export class UnauthorizedError extends AppError {
  constructor(message = "Sesi berakhir, silakan login lagi.") {
    super(message, 401);
  }
}

export class NotFoundError extends AppError {
  constructor(message = "Data tidak ditemukan.") {
    super(message, 404);
  }
}

export class ConflictError extends AppError {
  constructor(message: string) {
    super(message, 409);
  }
}

export class TooManyRequestsError extends AppError {
  constructor(message: string) {
    super(message, 429);
  }

/** Bentuk body error JSON yang dibaca `apiFetch` di browser. */

  error: string;
  fieldErrors?: FieldErrors;
}

export function toErrorResponse(error: unknown): Response {
  if (error instanceof ValidationError) {
    return Response.json({ error: error.message, fieldErrors: error.fieldErrors } satisfies ErrorBody, { status: error.status });

  if (error instanceof AppError) {
    return Response.json({ error: error.message } satisfies ErrorBody, { status: error.status });
  }
  console.error(error);

import { z } from "zod";
import { ValidationError, type FieldErrors } from "./errors";

/** Validasi input tak tepercaya; gagal → ValidationError berisi pesan per field (error tanpa path → "_form"). */
export function parseOrThrow<S extends z.ZodType>(schema: S, input: unknown): z.output<S> {
  const result = schema.safeParse(input);
  if (result.success) return result.data;
  const flat = z.flattenError(result.error);
  const fieldErrors: FieldErrors = { ...(flat.fieldErrors as FieldErrors) };
  if (flat.formErrors.length) fieldErrors._form = flat.formErrors;
  throw new ValidationError(fieldErrors);
}

  const result = schema.safeParse(source);
  if (!result.success) {

    throw new Error(`Env tidak valid — ${detail}`);

// Token sesi: base64url(JSON {exp}) + "." + base64url(HMAC-SHA256). Hanya Web Crypto → bisa dipakai di proxy & route.
export const SESSION_COOKIE = "udp_session";
export const SESSION_MAX_AGE = 60 * 60 * 24 * 30; // detik (30 hari)

const encoder = new TextEncoder();

function toBase64Url(bytes: Uint8Array): string {
  let binary = "";
  for (const byte of bytes) binary += String.fromCharCode(byte);
  return btoa(binary).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
}

function fromBase64Url(value: string): Uint8Array<ArrayBuffer> {
  const binary = atob(value.replace(/-/g, "+").replace(/_/g, "/"));
  return Uint8Array.from(binary, (char) => char.charCodeAt(0));
}

function hmacKey(secret: string): Promise<CryptoKey> {
  return crypto.subtle.importKey("raw", encoder.encode(secret), { name: "HMAC", hash: "SHA-256" }, false, ["sign", "verify"]);
}

export async function createSessionToken(secret: string, now: number = Date.now()): Promise<string> {
  const payload = toBase64Url(encoder.encode(JSON.stringify({ exp: now + SESSION_MAX_AGE * 1000 })));
  const signature = await crypto.subtle.sign("HMAC", await hmacKey(secret), encoder.encode(payload));
  return `${payload}.${toBase64Url(new Uint8Array(signature))}`;
}

export async function verifySessionToken(token: string | undefined, secret: string, now: number = Date.now()): Promise<boolean> {
  if (!token || !secret) return false;
  const [payload, signature] = token.split(".");
  if (!payload || !signature) return false;
  try {
    const valid = await crypto.subtle.verify("HMAC", await hmacKey(secret), fromBase64Url(signature), encoder.encode(payload));
    if (!valid) return false;
    const { exp } = JSON.parse(new TextDecoder().decode(fromBase64Url(payload))) as { exp?: unknown };
    return typeof exp === "number" && exp > now;
  } catch {
    return false;
  }
}

import "server-only";
import { verify } from "@node-rs/argon2";
import { z } from "zod";
import type { Db } from "@/db/types";
import { TooManyRequestsError, ValidationError } from "@/lib/errors";
import { parseOrThrow } from "@/lib/validation";
import { countFailedAttemptsSince, insertLoginAttempt } from "../api/login-attempts-repo";

export const MAX_FAILED_ATTEMPTS = 5;
const LOCKOUT_WINDOW_MS = 15 * 60 * 1000;

const loginSchema = z.object({ password: z.string({ error: "Password wajib diisi." }).min(1, "Password wajib diisi.") });

/** Cek password aplikasi (spec §7). Gagal → ValidationError; terlalu banyak gagal per IP → TooManyRequestsError. */
export async function login(db: Db, input: unknown, context: { ip: string; passwordHash: string; now?: Date }): Promise<void> {
  const { password } = parseOrThrow(loginSchema, input);
  const now = context.now ?? new Date();
  const failures = await countFailedAttemptsSince(db, context.ip, new Date(now.getTime() - LOCKOUT_WINDOW_MS));
  if (failures >= MAX_FAILED_ATTEMPTS) {
    throw new TooManyRequestsError("Terlalu banyak percobaan gagal. Coba lagi 15 menit lagi.");
  }
  const ok = await verify(context.passwordHash, password).catch(() => false);
  await insertLoginAttempt(db, { ip: context.ip, success: ok, attemptedAt: now });
  if (!ok) throw new ValidationError({ password: ["Password salah."] }, "Password salah.");
}

    return await request.json();
  } catch {
    throw new ValidationError({ _form: ["Body request harus JSON."] }, "Body request harus JSON.");
  }

  return request.headers.get("x-forwarded-for")?.split(",")[0]?.trim() || "unknown";

    try {
      if (!options.public) await requireSession();
      return await handler(request, context);
    } catch (error) {

    }

import { NextResponse, type NextRequest } from "next/server";
import { SESSION_COOKIE, verifySessionToken } from "@/lib/session";

const PUBLIC_PATHS = new Set(["/login", "/api/auth/login", "/api/auth/logout", "/api/auth/google/callback"]);

export async function proxy(request: NextRequest) {
  const { pathname, search } = request.nextUrl;
  if (PUBLIC_PATHS.has(pathname)) return NextResponse.next();

  const token = request.cookies.get(SESSION_COOKIE)?.value;
  if (await verifySessionToken(token, process.env.SESSION_SECRET ?? "")) return NextResponse.next();

  if (pathname.startsWith("/api/")) {
    return Response.json({ error: "Sesi berakhir, silakan login lagi." }, { status: 401 });
  }
  const loginUrl = new URL("/login", request.url);
  loginUrl.searchParams.set("next", pathname + search);
  return NextResponse.redirect(loginUrl);
}

```ts
// Salin komponen NeoBrutalism (ex-RetroUI) apa adanya dari registry ke src/components/ui.
// `shadcn add` me-resolve registryDependencies tanpa namespace (mis. "button") ke registry default shadcn,
// sehingga versi NeoBrutalism tertimpa (terbukti 2026-10-21) — script ini me-resolve semuanya dari registry NeoBrutalism.
// `npm run ui:sync`  → tulis file + tampilkan paket npm yang dibutuhkan
// `npm run ui:check` → gagal jika ada file yang hilang/diedit (komponen asli tidak boleh diubah)
import { mkdir, readFile, writeFile } from "node:fs/promises";
import path from "node:path";

interface RegistryItem {
  name: string;
  dependencies?: string[];
  registryDependencies?: string[];
  files: { path: string; content: string }[];
}

interface UiConfig {
  registry: string;
  outDir: string;
  components: string[];
}

const config = JSON.parse(await readFile("ui.config.json", "utf8")) as UiConfig;
const check = process.argv.includes("--check");

async function fetchItem(name: string): Promise<RegistryItem> {
  const res = await fetch(config.registry.replace("{name}", name));
  if (!res.ok) throw new Error(`Registry item "${name}" gagal diambil: HTTP ${res.status}`);
  return (await res.json()) as RegistryItem;
}

const items = new Map<string, RegistryItem>();
const queue = [...config.components];
while (queue.length) {
  const name = queue.shift()!;
  if (items.has(name)) continue;
  const item = await fetchItem(name);
  items.set(name, item);
  queue.push(...(item.registryDependencies ?? []));
}

const npmDeps = new Set<string>();
const problems: string[] = [];
for (const item of items.values()) {
  item.dependencies?.forEach((dep) => npmDeps.add(dep));
  for (const file of item.files) {
    const target = path.join(config.outDir, path.basename(file.path));
    if (check) {
      const local = await readFile(target, "utf8").catch(() => null);
      if (local === null) problems.push(`hilang: ${target}`);
      else if (local !== file.content) problems.push(`berbeda dari registry: ${target}`);
    } else {
      await mkdir(config.outDir, { recursive: true });
      await writeFile(target, file.content);
    }
  }
}

if (check) {
  if (problems.length) {
    console.error(problems.join("\n"));
    console.error("Jangan edit src/components/ui — bungkus di src/components/app. Jalankan `npm run ui:sync` untuk memulihkan.");
    process.exit(1);
  }
  console.log(`OK: ${items.size} komponen sama dengan registry.`);
} else {
  console.log(`Ditulis: ${[...items.keys()].sort().join(", ")}`);
  console.log(`Paket npm yang dibutuhkan: npm i ${[...npmDeps].sort().join(" ")}`);
}
```

import type { ErrorBody, FieldErrors } from "./errors";

export class ApiError extends Error {
  readonly status: number;
  readonly fieldErrors: FieldErrors;
  constructor(status: number, message: string, fieldErrors: FieldErrors = {}) {
    super(message);
    this.name = "ApiError";
    this.status = status;
    this.fieldErrors = fieldErrors;
  }
}

/** Panggil /api/* dari browser. Error → ApiError (pesan + error per field). 401 → ke halaman login. */
export async function apiFetch<T>(url: string, init: Omit<RequestInit, "body"> & { json?: unknown } = {}): Promise<T> {
  const { json, headers, ...rest } = init;
  const response = await fetch(url, {
    ...rest,
    headers: { ...(json !== undefined ? { "content-type": "application/json" } : {}), ...headers },
    body: json !== undefined ? JSON.stringify(json) : undefined,
  });
  if (response.status === 401 && typeof window !== "undefined") {
    window.location.href = `/login?next=${encodeURIComponent(window.location.pathname + window.location.search)}`;
  }
  if (!response.ok) {
    const body = (await response.json().catch(() => ({}))) as Partial<ErrorBody>;
    throw new ApiError(response.status, body.error ?? `Permintaan gagal (HTTP ${response.status}).`, body.fieldErrors);
  }
  if (response.status === 204) return undefined as T;
  return (await response.json()) as T;
}

import { useMutation } from "@tanstack/react-query";
import { useRouter, useSearchParams } from "next/navigation";
import { useState } from "react";
import { Button } from "@/components/ui/button";
import { Field, FieldError, FieldLabel } from "@/components/ui/field";
import { Input } from "@/components/ui/input";
import { ApiError, apiFetch } from "@/lib/api-client";
import { safeNextPath } from "../lib/safe-next";

export function LoginForm() {
  const router = useRouter();
  const next = safeNextPath(useSearchParams().get("next"));
  const [password, setPassword] = useState("");
  const mutation = useMutation({
    mutationFn: () => apiFetch<void>("/api/auth/login", { method: "POST", json: { password } }),
    onSuccess: () => {
      router.replace(next);
      router.refresh();
    },
  });
  const error = mutation.error instanceof ApiError ? (mutation.error.fieldErrors.password?.[0] ?? mutation.error.message) : null;

  return (
    <form
      className="flex flex-col gap-4"
      onSubmit={(event) => {
        event.preventDefault();
        mutation.mutate();
      }}
    >
      <Field data-invalid={error ? true : undefined}>
        <FieldLabel htmlFor="password">Password</FieldLabel>
        <Input
          id="password"
          type="password"
          autoComplete="current-password"
          autoFocus
          value={password}
          onChange={(event) => setPassword(event.target.value)}
          aria-invalid={error ? true : undefined}
        />
        {error && <FieldError>{error}</FieldError>}
      </Field>
      <Button type="submit" size="lg" disabled={mutation.isPending || !password}>
        {mutation.isPending ? "Memeriksa…" : "Masuk"}
      </Button>
    </form>
  );
}

import "server-only";
import { z } from "zod";
import type { Db } from "@/db/types";
import { ConflictError, NotFoundError, ValidationError } from "@/lib/errors";
import { parseOrThrow } from "@/lib/validation";
import { selectProject, type NewProjectRow, type ProjectRow } from "../api/projects-repo";
import { createProjectSchema, PROJECT_END_BEFORE_START, updateProjectSchema } from "../types/project-input";

export const PROJECT_LOCKED_MESSAGE = "Proyek sudah selesai dan dikunci. Buka kembali proyek di Pengaturan untuk mengubah data.";

export function parseCreateProject(input: unknown) {
  return parseOrThrow(createProjectSchema, input);
}

/** Proyek dari id (uuid tidak valid diperlakukan sebagai tidak ditemukan, bukan error DB). */
export async function findProjectOrThrow(db: Db, id: string): Promise<ProjectRow> {
  const row = z.uuid().safeParse(id).success ? await selectProject(db, id) : undefined;
  if (!row) throw new NotFoundError("Proyek tidak ditemukan.");
  return row;
}

/** Aturan bisnis spec §3.4: proyek `done` menolak semua mutasi. Dipakai juga oleh command modul lain. */
export function assertProjectEditable(row: ProjectRow): void {
  if (row.status === "done") throw new ConflictError(PROJECT_LOCKED_MESSAGE);
}

/** Validasi perubahan proyek terhadap data lama; proyek selesai hanya boleh diubah statusnya. */
export function planProjectUpdate(existing: ProjectRow, input: unknown): Partial<NewProjectRow> {
  const patch = parseOrThrow(updateProjectSchema, input);
  const onlyStatus = Object.keys(patch).every((key) => key === "status");
  if (!onlyStatus) assertProjectEditable(existing);
  const start = patch.startDate ?? existing.startDate;
  const end = patch.endDate ?? existing.endDate;
  if (end < start) throw new ValidationError({ endDate: [PROJECT_END_BEFORE_START] });
  return patch;
}

import type { Db } from "@/db/types";
import { insertBudget, insertBudgetCategories } from "@/modules/budgets/api/budgets-repo";
import { insertCategories } from "@/modules/budgets/api/categories-repo";
import { insertWallets } from "@/modules/wallets/api/wallets-repo";
import { insertProject } from "../api/projects-repo";
import { insertVehicles } from "../api/vehicles-repo";
import { buildProjectDefaults } from "../lib/seed-defaults";
import { toProject } from "../mappers/project";
import { parseCreateProject } from "../services/projects";
import type { Project } from "../types/project";

/** Buat proyek + seed default (spec §3.2) dalam satu transaksi: gagal di tengah → tidak ada yang tersimpan. */
export async function createProject(db: Db, input: unknown): Promise<Project> {
  const data = parseCreateProject(input);
  return db.transaction(async (tx) => {
    const project = await insertProject(tx, {
      name: data.name,
      location: data.location,
      startDate: data.startDate,
      endDate: data.endDate,
      pax: data.pax,
    });
    const vehicleRows = await insertVehicles(
      tx,
      data.vehicles.map((name, index) => ({ projectId: project.id, name, sortOrder: index })),
    );
    const defaults = buildProjectDefaults(vehicleRows.map((v) => ({ id: v.id, name: v.name })));
    await insertWallets(
      tx,
      defaults.wallets.map((wallet, index) => ({ ...wallet, projectId: project.id, sortOrder: index })),
    );
    const categoryRows = await insertCategories(
      tx,
      defaults.categories.map((category, index) => ({ ...category, projectId: project.id, sortOrder: index })),
    );
    for (const seed of defaults.budgets) {
      const budget = await insertBudget(tx, { projectId: project.id, name: seed.name, rate: seed.rate, basis: seed.basis, note: seed.note });
      const ids = categoryRows.filter((c) => seed.categoryNames.includes(c.name)).map((c) => c.id);
      await insertBudgetCategories(tx, budget.id, ids);
    }
    return toProject(project);
  });
}

# Alur data global

Kontrak ini diadaptasi dari `srikandi-merdeka-cup/docs/architecture/data-flow.md` dan menjadi sumber kebenaran untuk alur data dan kepemilikan folder di `src/`. Bagian "Adaptasi" mencatat pengecualian yang disengaja beserta alasannya.

## Arah dependensi

Baca (render server):

```text
app route → module index → view (komposisi + Suspense) → section (batas async server)
  → query (orkestrasi + dedupe) → service (validasi + aturan bisnis) → api (adapter/repo) → Neon / API eksternal
```

Baca (refresh dari browser):

```text
hook (useQuery) → app/api route (parse HTTP + sesi) → query yang SAMA → service → api
```

Tulis:

```text
form → hook (useMutation) → app/api route (parse HTTP + sesi) → command (db.transaction) → service → api (repo) → Neon
onSuccess → invalidate query key terkait
```

Jangan melompati layer:

- Komponen tidak memanggil DB/API eksternal.
- Route `app/api` tidak menduplikasi logika service/domain.
- Adapter/repo tidak membuat model UI.
- Row DB / respons eksternal mentah tidak sampai ke view atau komponen.
- Layer bawah tidak mengimpor dari layer render/routing.

## Kontrak folder

| Folder/file | Pemilik | Aturan |
| --- | --- | --- |
| `src/app/**/page.tsx` | entri route | Tipis: teruskan ke module. |
| `src/app/api/**/route.ts` | batas HTTP browser | Bungkus dengan `route()`; parse input, panggil query/command, serialisasi hasil. |
| `src/modules/*/index.tsx` | batas publik module | Delegasi ke view. Tanpa fetch/validasi/mapping. |
| `views/` | komposisi halaman + Suspense | Tanpa transport/normalisasi. |
| `sections/` | batas async server | Panggil query, prefetch ke QueryClient + `HydrationBoundary`, pilih state missing. |
| `queries/` | orkestrasi baca (server-only) | React `cache()` untuk dedupe per request. |
| `commands/` | orkestrasi tulis (server-only) | Satu use case mutasi dalam satu `db.transaction`. |
| `services/` | use case + validasi (server-only) | zod (`parseOrThrow`), aturan bisnis, panggil mapper. |
| `api/` | adapter transport (server-only) | Repo Drizzle per tabel & adapter eksternal; kembalikan row/`unknown` mentah. |
| `mappers/` | batas kepercayaan | Row/JSON mentah → tipe kanonik. |
| `lib/` | logika domain murni | Tanpa network, DB, state React, API browser. Wajib diuji. |
| `types/` | kontrak kanonik | Tipe + skema zod input yang client-safe. |
| `hooks/` | client-only | Hook TanStack Query + query key factory; hanya memanggil `/api/*`. |
| `components/` | render + interaksi lokal | `'use client'` hanya jika perlu. |
| `src/components/ui/` | NeoBrutalism | **Tidak pernah diedit**; dijaga `npm run ui:check`. |
| `src/components/app/` | komponen app lintas module | Membungkus/meng-compose `ui`. |

## Adaptasi dari kontrak srikandi

1. **`commands/`**: kontrak asal hanya punya alur baca. App ini sebagian besar menulis, jadi `commands/` menjadi pasangan `queries/`.
2. **`api/` mencakup repo DB**: fungsi Drizzle diperlakukan sebagai adapter transport, setara dengan adapter API eksternal (Gemini, Google, Blob).
3. **Cache**: tidak ada `'use cache'` persisten untuk data keuangan karena data harus segar setelah mutasi. Otoritas kesegaran adalah DB. React `cache()` hanya untuk dedupe per request, dan cache browser dipegang TanStack Query (invalidate pada `onSuccess`). Karena `cacheComponents` aktif, setiap akses DB/cookies di halaman berada di dalam `<Suspense>`.
4. **`hooks/`**: folder client-only untuk TanStack Query. Query key factory berada di `hooks/keys.ts` agar section server dapat mem-prefetch dengan key yang sama.

## Batas runtime

- `queries/`, `commands/`, `services/`, `api/`, `src/db/client.ts`, `src/lib/env.ts`, dan `src/lib/http.ts` mengimpor `server-only`.
- `src/lib/session.ts` sengaja tanpa `server-only`/`Buffer` karena dipakai `src/proxy.ts`.
- Fungsi server menerima `db: Db` sebagai parameter pertama. Produksi memakai `getDb()` (Neon), test memakai `createTestDb()` (PGlite).

## Missing, empty, failed

- **Missing:** `NotFoundError` → 404 (API) / `notFound()` (halaman).
- **Empty:** data ada tapi kosong → empty state dengan CTA.
- **Failed:** error DB/eksternal → error boundary per section / respons 500. Jangan diubah menjadi data kosong.

## Checklist perubahan

1. Tentukan kontrak sumber, tipe kanonik, pemilik cache, dan semua konsumen.
2. Ubah di layer yang tepat (transport, validasi, mapping, domain, render).
3. Pertahankan jalur query yang sama untuk render server dan endpoint browser.
4. Uji missing/empty/failed secara terpisah.
5. Tambah/ubah test di batas yang berubah.
