> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Pagu mengikuti surat penawaran dan bisa diedit dari halaman **Lainnya → Pagu**: tarif, cara hitung, dan kategori yang dihitung bisa diubah, pagu bisa ditambah atau dihapus (dengan konfirmasi), serta ada tombol untuk menerapkan pagu default ke proyek lama.

**Surat penawaran (default proyek baru):**

| Pagu | Tarif | Cara hitung | Kategori |
| --- | --- | --- | --- |
| Pagu Makan | Rp150.000 | per orang per hari | Makan Siang, Makan Malam |
| Pagu Sarapan | Rp25.000 | per orang per hari | Sarapan |
| BBM + Tol | Rp1.300.000 | **per mobil per trip** (basis baru `per_vehicle`, tidak dikali hari) | BBM Perjalanan, Tol |
| Operasional Harian | Rp100.000 | per mobil per hari | Bensin Harian, Parkir, Lain-lain |

Kategori "BBM" dipecah menjadi **BBM Perjalanan** (masuk pagu BBM + Tol) dan **Bensin Harian** (masuk Operasional Harian). Keputusan pengguna 2026-10-21.

**Architecture:** Sama dengan Rencana 2a. Budget CRUD: route → command → `planBudget` (zod + aturan) → repo. Halaman pengaturan memakai query `getBudgetSettings` (dari `getLedger`) yang di-hydrate ke TanStack Query di bawah `ledgerKeys.all(projectId)`, jadi beranda dan kalender ikut segar setelah pagu diubah.

/** Batas per hari; null untuk basis yang tidak dikali hari (per_vehicle, lump_sum). */
export function dailyLimit(budget: Pick<Budget, "rate" | "basis">, ctx: BudgetContext): number | null {
  switch (budget.basis) {
    case "per_pax_per_day":
      return budget.rate * ctx.pax;
    case "per_vehicle_per_day":
      return budget.rate * ctx.vehicleCount;
    case "per_day":
      return budget.rate;
    case "per_vehicle":
    case "lump_sum":
      return null;
  }
}

export function budgetLimit(budget: Pick<Budget, "rate" | "basis">, ctx: BudgetContext): number {
  if (budget.basis === "per_vehicle") return budget.rate * ctx.vehicleCount;
  const daily = dailyLimit(budget, ctx);
  return daily === null ? budget.rate : daily * ctx.days;
}

import type { BudgetContext } from "./budget";

  per_pax_per_day: "per orang per hari",
  per_vehicle_per_day: "per mobil per hari",
  per_vehicle: "per mobil per trip",
  per_day: "per hari",
  lump_sum: "total sekali",
};

/** "Rp 150.000 × 9 orang × 6 hari" — menjelaskan asal angka pagu. */
export function limitFormula(budget: Pick<Budget, "rate" | "basis">, ctx: BudgetContext): string {
  const rate = formatRupiah(budget.rate);
  switch (budget.basis) {
    case "per_pax_per_day":
      return `${rate} × ${ctx.pax} orang × ${ctx.days} hari`;
    case "per_vehicle_per_day":
      return `${rate} × ${ctx.vehicleCount} mobil × ${ctx.days} hari`;
    case "per_vehicle":
      return `${rate} × ${ctx.vehicleCount} mobil`;
    case "per_day":
      return `${rate} × ${ctx.days} hari`;
    case "lump_sum":
      return rate;
  }
}

import { z } from "zod";
import type { Db } from "@/db/types";
import { NotFoundError, ValidationError, type FieldErrors } from "@/lib/errors";
import { parseOrThrow } from "@/lib/validation";
import { selectBudget, type BudgetRow } from "../api/budgets-repo";
import type { Category } from "../types/category";

export interface BudgetValues {
  name: string;
  rate: number;
  basis: BudgetRow["basis"];
  note: string | null;
  categoryIds: string[];
}

/** Validasi pagu: nama unik di proyek, kategori = pengeluaran milik proyek. */
export function planBudget(input: unknown, refs: { categories: Category[]; otherBudgetNames: string[] }): BudgetValues {
  const v = parseOrThrow(budgetInputSchema, input);
  const errors: FieldErrors = {};
  if (refs.otherBudgetNames.some((n) => n.toLowerCase() === v.name.toLowerCase())) errors.name = ["Nama pagu sudah dipakai."];
  const byId = new Map(refs.categories.map((c) => [c.id, c]));
  const picked = v.categoryIds.map((id) => byId.get(id));
  if (picked.some((c) => !c)) errors.categoryIds = ["Kategori tidak ditemukan di proyek ini."];
  else if (picked.some((c) => c!.kind !== "expense")) errors.categoryIds = ["Pagu hanya untuk kategori pengeluaran."];
  if (Object.keys(errors).length) throw new ValidationError(errors);
  return { ...v, categoryIds: [...new Set(v.categoryIds)] };
}

export async function findBudgetOrThrow(db: Db, projectId: string, id: string): Promise<BudgetRow> {
  const row = z.uuid().safeParse(id).success ? await selectBudget(db, projectId, id) : undefined;
  if (!row) throw new NotFoundError("Pagu tidak ditemukan.");

async function refs(db: Db, projectId: string, exceptBudgetId?: string) {
  const [categoryRows, budgetRows] = await Promise.all([selectCategories(db, projectId), selectBudgets(db, projectId)]);
  return { categories: categoryRows.map(toCategory), otherBudgetNames: budgetRows.filter((b) => b.id !== exceptBudgetId).map((b) => b.name) };

export async function createBudget(db: Db, projectId: string, input: unknown): Promise<Budget> {
  await editableProject(db, projectId);
  const { categoryIds, ...values } = planBudget(input, await refs(db, projectId));
  return db.transaction(async (tx) => {
    const row = await insertBudget(tx, { ...values, projectId });
    await insertBudgetCategories(tx, row.id, categoryIds);
    return toBudget(row, categoryIds);
  });
}

export async function updateBudget(db: Db, projectId: string, id: string, input: unknown): Promise<Budget> {
  await editableProject(db, projectId);
  await findBudgetOrThrow(db, projectId, id);
  const { categoryIds, ...values } = planBudget(input, await refs(db, projectId, id));
  return db.transaction(async (tx) => {
    const row = await updateBudgetRow(tx, id, values);
    await replaceBudgetCategories(tx, id, categoryIds);
    return toBudget(row, categoryIds);
  });

export interface ApplyDefaultsResult {
  renamed: string[];
  addedCategories: string[];
  addedBudgets: string[];
}

/**
 * Lengkapi proyek dengan kategori & pagu default (surat penawaran) yang belum ada — berdasarkan nama.
 * Tidak mengubah tarif/kategori pagu yang sudah ada. Proyek lama: "BBM" → "BBM Perjalanan".
 */
export async function applyDefaultBudgets(db: Db, projectId: string): Promise<ApplyDefaultsResult> {
  const project = await editableProject(db, projectId);
  return db.transaction(async (tx) => {
    const result: ApplyDefaultsResult = { renamed: [], addedCategories: [], addedBudgets: [] };
    const defaults = buildProjectDefaults([]);
    let categoryRows = await selectCategories(tx, project.id);
    const byName = () => new Map(categoryRows.map((c) => [c.name, c]));

    const legacyFuel = byName().get("BBM");
    if (legacyFuel && !byName().has("BBM Perjalanan")) {
      await updateCategoryRow(tx, legacyFuel.id, { name: "BBM Perjalanan" });
      result.renamed.push("BBM → BBM Perjalanan");
    }

    categoryRows = await selectCategories(tx, project.id);
    const missing = defaults.categories.filter((c) => !byName().has(c.name));
    await insertCategories(
      tx,
      missing.map((c) => ({ ...c, projectId: project.id, sortOrder: defaults.categories.indexOf(c) })),
    );
    result.addedCategories.push(...missing.map((c) => c.name));
    categoryRows = await selectCategories(tx, project.id);
    // urutkan kategori default sesuai urutan surat penawaran; kategori buatan pengguna di belakang
    for (const row of categoryRows) {
      const index = defaults.categories.findIndex((c) => c.name === row.name);
      const sortOrder = index === -1 ? defaults.categories.length + row.sortOrder : index;
      if (row.sortOrder !== sortOrder) await updateCategoryRow(tx, row.id, { sortOrder });
    }

    const existingBudgets = new Set((await selectBudgets(tx, project.id)).map((b) => b.name));
    const ids = byName();
    for (const seed of defaults.budgets.filter((b) => !existingBudgets.has(b.name))) {
      const row = await insertBudget(tx, { projectId: project.id, name: seed.name, rate: seed.rate, basis: seed.basis, note: seed.note });
      await insertBudgetCategories(
        tx,
        row.id,
        seed.categoryNames.map((n) => ids.get(n)?.id).filter((id): id is string => !!id),
      );
      result.addedBudgets.push(seed.name);
    }
    return result;
  });
}

/** Pagu memengaruhi beranda & kalender → invalidasi seluruh data buku kas proyek. */
function useLedgerMutation<TInput, TResult>(projectId: string, fn: (input: TInput) => Promise<TResult>) {
  const queryClient = useQueryClient();
  return useMutation({ mutationFn: fn, onSuccess: () => queryClient.invalidateQueries({ queryKey: ledgerKeys.all(projectId) }) });
}

export function useSaveBudget(projectId: string, budgetId?: string) {
  return useLedgerMutation(projectId, (input: BudgetInput) =>
    budgetId ? apiFetch<Budget>(`${base(projectId)}/${budgetId}`, { method: "PATCH", json: input }) : apiFetch<Budget>(base(projectId), { method: "POST", json: input }),
  );
}

export function useDeleteBudget(projectId: string) {
  return useLedgerMutation(projectId, (budgetId: string) => apiFetch<void>(`${base(projectId)}/${budgetId}`, { method: "DELETE" }));
}

export function useApplyDefaultBudgets(projectId: string) {
  return useLedgerMutation(projectId, () => apiFetch<ApplyDefaultsResult>(`${base(projectId)}/defaults`, { method: "POST" }));

import { BudgetSettingsList } from "../components/budget-settings";
import { getBudgetSettings } from "../queries/budget-settings";

export async function BudgetSettingsSection() {
  await connection();
  const db = getDb();
  const project = await getActiveProject(db);
  if (!project) redirect("/proyek/baru");
  const queryClient = new QueryClient();
  queryClient.setQueryData(ledgerKeys.budgets(project.id), await getBudgetSettings(db, project.id));
  return (
    <HydrationBoundary state={dehydrate(queryClient)}>
      <BudgetSettingsList projectId={project.id} />
    </HydrationBoundary>
  );
}
