## Aturan umum

Sama dengan rencana sebelumnya. Tambahan:

- **Aplikasi generik:** tidak ada kategori/angka khusus proyek di kode. Ikon kategori dipilih dari daftar ikon umum.
- **Kategori sistem** (`isAdminFee` untuk biaya admin transfer, `isOfficeFunding` untuk UDP kantor) boleh diganti nama, tidak boleh diarsipkan. Jenis kategori (pengeluaran/pemasukan) tidak bisa diubah setelah dibuat.
- **Mobil boleh dihapus** jika tidak ada transaksi aktif yang memakainya dan dompet Flazz-nya kosong (tanpa transaksi apa pun termasuk yang dihapus — FK `RESTRICT` — dan saldo awal 0). Dompet Flazz kosong itu ikut dihapus. Ganti nama mobil juga mengganti nama dompet "Flazz {nama lama}".
- **Hapus proyek:** tombol dua langkah + password admin. Proyek selesai pun bisa dihapus. Setelah sukses, reload penuh ke `/proyek` (cache & state halaman dibuang).
- Maksimal 5 mobil per proyek (sama dengan form proyek baru).

```ts
import "server-only";
import { z } from "zod";
import type { Db } from "@/db/types";
import { NotFoundError, ValidationError, type FieldErrors } from "@/lib/errors";
import { parseOrThrow } from "@/lib/validation";
import type { CategoryKind } from "@/modules/shared/types/enums";
import { selectCategory, type CategoryRow } from "../api/categories-repo";
import { categoryInputSchema } from "../types/category-input";

export interface CategoryValues {
  kind: CategoryKind;
  name: string;
  icon: string;
  isVehicleRelated: boolean;
  archived: boolean;
}

/** Nama unik di proyek; jenis tetap; terkait mobil hanya pengeluaran; kategori sistem tidak bisa diarsipkan. */
export function planCategory(input: unknown, refs: { existing?: CategoryRow; otherNames: string[] }): CategoryValues {
  const v = parseOrThrow(categoryInputSchema, input);
  const errors: FieldErrors = {};
  const { existing } = refs;
  if (refs.otherNames.some((n) => n.toLowerCase() === v.name.toLowerCase())) errors.name = ["Nama kategori sudah dipakai."];
  if (existing && existing.kind !== v.kind) errors.kind = ["Jenis kategori tidak bisa diubah."];
  if (v.isVehicleRelated && v.kind !== "expense") errors.isVehicleRelated = ["Hanya kategori pengeluaran yang bisa terkait mobil."];
  if (existing?.isAdminFee && v.archived) errors.archived = ["Kategori biaya admin dipakai otomatis untuk transfer, tidak bisa diarsipkan."];
  if (existing?.isOfficeFunding && v.archived) errors.archived = ["Kategori UDP kantor dipakai untuk menghitung UDP diterima, tidak bisa diarsipkan."];
  if (Object.keys(errors).length) throw new ValidationError(errors);
  return v;
}

export async function findCategoryOrThrow(db: Db, projectId: string, id: string): Promise<CategoryRow> {
  const row = z.uuid().safeParse(id).success ? await selectCategory(db, projectId, id) : undefined;
  if (!row) throw new NotFoundError("Kategori tidak ditemukan.");
  return row;
}
```

`src/modules/budgets/commands/save-category.ts`:

```ts
import "server-only";
import type { Db } from "@/db/types";
import { assertProjectEditable, findProjectOrThrow } from "@/modules/projects/services/projects";
import { insertCategories, selectCategories, updateCategoryRow } from "../api/categories-repo";
import { toCategory } from "../mappers/category";
import { findCategoryOrThrow, planCategory } from "../services/categories";
import type { Category } from "../types/category";

async function editableProject(db: Db, projectId: string) {
  assertProjectEditable(await findProjectOrThrow(db, projectId));
}

async function refs(db: Db, projectId: string, exceptId?: string) {
  const rows = await selectCategories(db, projectId);
  return {
    otherNames: rows.filter((r) => r.id !== exceptId).map((r) => r.name),
    nextSortOrder: Math.max(-1, ...rows.map((r) => r.sortOrder)) + 1,
  };
}

export async function createCategory(db: Db, projectId: string, input: unknown): Promise<Category> {
  await editableProject(db, projectId);
  const r = await refs(db, projectId);
  const values = planCategory(input, { otherNames: r.otherNames });
  const [row] = await insertCategories(db, [{ ...values, projectId, sortOrder: r.nextSortOrder }]);
  return toCategory(row);
}

export async function updateCategory(db: Db, projectId: string, id: string, input: unknown): Promise<Category> {
  await editableProject(db, projectId);
  const existing = await findCategoryOrThrow(db, projectId, id);
  const values = planCategory(input, { existing, otherNames: (await refs(db, projectId, id)).otherNames });
  return toCategory(await updateCategoryRow(db, id, values));
}
```

```ts
import "server-only";
import { cache } from "react";
import type { Db } from "@/db/types";
import { sumBy } from "@/modules/transactions/lib/ledger";
import { getLedger } from "@/modules/transactions/queries/ledger";
import type { Category } from "../types/category";

export interface CategorySetting extends Category {
  /** Jumlah transaksi aktif berkategori ini. */
  usedCount: number;
  /** Pagu yang menghitung kategori ini. */
  budgetNames: string[];
}

export interface CategorySettings {
  projectId: string;
  locked: boolean;
  /** Semua kategori termasuk yang diarsipkan, urut sortOrder. */
  categories: CategorySetting[];
}

export const getCategorySettings = cache(async (db: Db, projectId: string): Promise<CategorySettings> => {
  const ledger = await getLedger(db, projectId);
  const used = sumBy(
    ledger.transactions.filter((t) => t.categoryId),
    (t) => t.categoryId!,
    () => 1,
  );
  return {
    projectId,
    locked: ledger.project.status === "done",
    categories: ledger.categories.map((c) => ({
      ...c,
      usedCount: used.get(c.id) ?? 0,
      budgetNames: ledger.budgets.filter((b) => b.categoryIds.includes(c.id)).map((b) => b.name),
    })),
  };
});
```

```tsx
export function CategorySettingsList({ projectId }: { projectId: string }) {
  const { data } = useCategorySettings(projectId);

  const [showArchived, setShowArchived] = useState(false);
  if (!data) return null;

  const renderItem = (category: CategorySetting) =>

        </Alert>
      )}

        <Card key={kind}>
          <CardHeader>
            <CardTitle className="text-base">{title}</CardTitle>

                <CategoryForm projectId={projectId} defaultKind={kind} onDone={() => setEditing(null)} />
              </div>
            )}
            <ul className="divide-y">{data.categories.filter((c) => c.kind === kind && !c.archived).map(renderItem)}</ul>
          </CardContent>
        </Card>

function CategoryItem({ projectId, category, locked, onEdit }: { projectId: string; category: CategorySetting; locked: boolean; onEdit: () => void }) {

    category.isAdminFee && "biaya admin transfer",
    category.isOfficeFunding && "UDP kantor",
    category.usedCount > 0 && `${category.usedCount} transaksi`,

        {category.archived ? "Aktifkan" : "Arsipkan"}
      </Button>
    </li>
  );
}
```

```ts
export function flazzWalletName(vehicleName: string): string {
  return `Flazz ${vehicleName}`;
}

/** `base`, atau `base (2)`, `base (3)`, … bila nama sudah dipakai (tanpa beda huruf besar/kecil). */
export function uniqueName(base: string, taken: string[]): string {
  const used = new Set(taken.map((n) => n.toLowerCase()));
  if (!used.has(base.toLowerCase())) return base;
  for (let i = 2; ; i++) {
    const name = `${base} (${i})`;
    if (!used.has(name.toLowerCase())) return name;
  }
}

interface RemovalRefs {
  wallets: { id: string; vehicleId: string | null; openingBalance: number }[];
  /** Semua transaksi proyek termasuk yang dihapus (FK transaksi → dompet RESTRICT). */
  transactions: { vehicleId: string | null; walletId: string; toWalletId: string | null; deletedAt: Date | null }[];
}

/** Mobil boleh dihapus jika tidak dipakai transaksi aktif dan dompet Flazz-nya kosong; dompet itu ikut dihapus. */
export function vehicleRemoval(vehicleId: string, refs: RemovalRefs): { allowed: boolean; walletIds: string[] } {
  const walletIds = refs.wallets.filter((w) => w.vehicleId === vehicleId).map((w) => w.id);
  const ids = new Set(walletIds);
  const usedByTransaction = refs.transactions.some((t) => !t.deletedAt && t.vehicleId === vehicleId);
  const walletUsed =
    refs.wallets.some((w) => ids.has(w.id) && w.openingBalance !== 0) ||
    refs.transactions.some((t) => ids.has(t.walletId) || (t.toWalletId !== null && ids.has(t.toWalletId)));
  return { allowed: !usedByTransaction && !walletUsed, walletIds };
}
```

```ts
const nextSortOrder = (rows: { sortOrder: number }[]) => Math.max(-1, ...rows.map((r) => r.sortOrder)) + 1;

/** Mobil baru + dompet "Flazz {mobil}" (keputusan pengguna 2026-10-21). */
export async function createVehicle(db: Db, projectId: string, input: unknown): Promise<Vehicle> {
  await editableProject(db, projectId);
  const [vehicleRows, walletRows] = await Promise.all([selectVehicles(db, projectId), selectWallets(db, projectId)]);
  const values = planVehicle(input, { otherNames: vehicleRows.map((v) => v.name), adding: true });
  return db.transaction(async (tx) => {
    const [vehicle] = await insertVehicles(tx, [{ ...values, projectId, sortOrder: nextSortOrder(vehicleRows) }]);
    await insertWallets(tx, [
      {
        projectId,
        kind: "flazz",
        name: uniqueName(flazzWalletName(vehicle.name), walletRows.map((w) => w.name)),
        vehicleId: vehicle.id,
        sortOrder: nextSortOrder(walletRows),
      },
    ]);
    return toVehicle(vehicle);
  });
}

/** Ganti nama/plat. Dompet bernama "Flazz {nama lama}" milik mobil ini ikut diganti namanya. */
export async function updateVehicle(db: Db, projectId: string, id: string, input: unknown): Promise<Vehicle> {
  await editableProject(db, projectId);
  const existing = await findVehicleOrThrow(db, projectId, id);
  const [vehicleRows, walletRows] = await Promise.all([selectVehicles(db, projectId), selectWallets(db, projectId)]);
  const values = planVehicle(input, { otherNames: vehicleRows.filter((v) => v.id !== id).map((v) => v.name), adding: false });
  return db.transaction(async (tx) => {
    const row = await updateVehicleRow(tx, id, values);
    if (values.name !== existing.name) {
      for (const wallet of walletRows.filter((w) => w.vehicleId === id && w.name === flazzWalletName(existing.name))) {
        const taken = walletRows.filter((w) => w.id !== wallet.id).map((w) => w.name);
        await updateWalletRow(tx, wallet.id, { name: uniqueName(flazzWalletName(values.name), taken) });
      }
    }
    return toVehicle(row);
  });
}

export async function deleteVehicle(db: Db, projectId: string, id: string): Promise<void> {
  await editableProject(db, projectId);
  await findVehicleOrThrow(db, projectId, id);
  const [walletRows, transactionRows] = await Promise.all([selectWallets(db, projectId), selectTransactionRowsIncludingDeleted(db, projectId)]);
  const removal = vehicleRemoval(id, { wallets: walletRows, transactions: transactionRows });
  if (!removal.allowed) {
    throw new ConflictError("Mobil sudah dipakai di transaksi atau dompet Flazz-nya sudah berisi. Ganti nama saja, atau arsipkan dompetnya.");
  }
  await db.transaction(async (tx) => {
    await deleteWalletRows(tx, removal.walletIds);
    await deleteVehicleRow(tx, id);
  });
}
```

```tsx
/** Dua langkah: buka → isi password admin → "Hapus permanen". Sukses → reload penuh agar cache & state halaman terbuang. */
export function DeleteProject({ project }: { project: ProjectDetail }) {
  const remove = useDeleteProject(project.id);
  const [open, setOpen] = useState(false);
  const [password, setPassword] = useState("");
  const fieldErrors = remove.error instanceof ApiError ? remove.error.fieldErrors : {};

  if (!open) {
    return (
      <Button variant="outline" className="text-destructive" onClick={() => setOpen(true)}>
        Hapus proyek…
      </Button>
    );
  }

  return (
    <form
      className="flex flex-col gap-3"
      onSubmit={(event) => {
        event.preventDefault();
        remove.mutate(password, {
          onSuccess: () => {
            // eslint-disable-next-line @next/next/no-location-assign-relative-destination
            window.location.href = "/proyek";
          },
        });
      }}
    >
      <p className="text-sm">
        Semua transaksi, dompet, kategori, pagu, cek saldo, dan foto bon proyek <span className="font-head">{project.name}</span> akan dihapus permanen dan tidak bisa dikembalikan.
      </p>
      <Field data-invalid={fieldErrors.password ? true : undefined}>
        <FieldLabel htmlFor="delete-password">Password admin</FieldLabel>
        <Input id="delete-password" type="password" autoComplete="current-password" className="h-10" value={password} onChange={(e) => setPassword(e.target.value)} />
        {fieldErrors.password && <FieldError>{fieldErrors.password[0]}</FieldError>}
      </Field>
      {remove.error && !fieldErrors.password && (
        <p role="alert" className="text-sm text-destructive">
          {remove.error.message}
        </p>
      )}
      <div className="flex gap-2">
        <Button type="submit" variant="destructive" className="flex-1" disabled={!password || remove.isPending}>
          {remove.isPending ? "Menghapus…" : "Hapus permanen"}
        </Button>
        <Button
          type="button"
          variant="outline"
          onClick={() => {
            setOpen(false);
            setPassword("");
          }}
        >
          Batal
        </Button>
      </div>
    </form>
  );
}
```

```tsx
  const { data } = useProjectSettings(projectId);
  if (!data) return null;
  const { project } = data;

      <Card>
        <CardHeader>

function ProjectDetailsForm({ project, locked }: { project: ProjectDetail; locked: boolean }) {
  const update = useUpdateProject(project.id);
  const [name, setName] = useState(project.name);
  const [location, setLocation] = useState(project.location);
  const [startDate, setStartDate] = useState(project.startDate);
  const [endDate, setEndDate] = useState(project.endDate);
  const [pax, setPax] = useState(String(project.pax));
  const fieldErrors = update.error instanceof ApiError ? update.error.fieldErrors : {};
  const errorOf = (field: string) => fieldErrors[field]?.[0];

  return (
    <form
      className="flex flex-col gap-4"
      onSubmit={(event) => {
        event.preventDefault();
        update.mutate({ name, location, startDate, endDate, pax: Number(pax) }, { onSuccess: () => toast.success("Pengaturan proyek disimpan.") });
      }}
    >
      <FieldGroup>
        <Field data-invalid={errorOf("name") ? true : undefined}>
          <FieldLabel htmlFor="settings-name">Nama proyek</FieldLabel>
          <Input id="settings-name" className="h-10" value={name} onChange={(e) => setName(e.target.value)} />
          {errorOf("name") && <FieldError>{errorOf("name")}</FieldError>}
        </Field>
        <Field data-invalid={errorOf("location") ? true : undefined}>
          <FieldLabel htmlFor="settings-location">Lokasi</FieldLabel>
          <Input id="settings-location" className="h-10" value={location} onChange={(e) => setLocation(e.target.value)} />
          {errorOf("location") && <FieldError>{errorOf("location")}</FieldError>}
        </Field>
        <div className="grid grid-cols-2 gap-3">
          <Field data-invalid={errorOf("startDate") ? true : undefined}>
            <FieldLabel htmlFor="settings-start">Mulai</FieldLabel>
            <Input id="settings-start" type="date" className="h-10" value={startDate} onChange={(e) => setStartDate(e.target.value)} />
            {errorOf("startDate") && <FieldError>{errorOf("startDate")}</FieldError>}
          </Field>
          <Field data-invalid={errorOf("endDate") ? true : undefined}>
            <FieldLabel htmlFor="settings-end">Selesai</FieldLabel>
            <Input id="settings-end" type="date" className="h-10" value={endDate} onChange={(e) => setEndDate(e.target.value)} />
            {errorOf("endDate") && <FieldError>{errorOf("endDate")}</FieldError>}
          </Field>
        </div>
        <Field data-invalid={errorOf("pax") ? true : undefined}>
          <FieldLabel htmlFor="settings-pax">Jumlah orang</FieldLabel>
          <Input id="settings-pax" className="h-10" inputMode="numeric" pattern="[0-9]*" value={pax} onChange={(e) => setPax(e.target.value.replace(/\D/g, ""))} />
          <FieldDescription>Periode dan jumlah orang dipakai untuk menghitung pagu.</FieldDescription>
          {errorOf("pax") && <FieldError>{errorOf("pax")}</FieldError>}
        </Field>
      </FieldGroup>
      {update.error && !Object.keys(fieldErrors).length && (
        <p role="alert" className="text-sm text-destructive">
          {update.error.message}
        </p>
      )}
      <Button type="submit" disabled={locked || update.isPending}>
        {update.isPending ? "Menyimpan…" : "Simpan"}
      </Button>
    </form>
  );
}

function ProjectStatus({ project }: { project: ProjectDetail }) {
  const update = useUpdateProject(project.id);
  const done = project.status === "done";
  return (
    <Button
      variant={done ? "default" : "outline"}
      className="w-full"
      disabled={update.isPending}
      onClick={() =>
        update.mutate(
          { status: done ? "active" : "done" },
          {
            onSuccess: () => toast.success(done ? "Proyek dibuka lagi." : "Proyek ditandai selesai dan dikunci."),
            onError: (error) => toast.error(error.message),
          },
        )
      }
    >
      {done ? "Buka lagi" : "Tandai selesai"}
    </Button>
  );
}
```

```tsx
import { dehydrate, HydrationBoundary, QueryClient } from "@tanstack/react-query";
import { redirect } from "next/navigation";
import { connection } from "next/server";
import { getDb } from "@/db/client";
import { ledgerKeys } from "@/modules/transactions/hooks/keys";
import { ProjectSettingsPanel } from "../components/project-settings";
import { getProjectSettings } from "../queries/project-settings";
import { getActiveProject } from "../queries/projects";

export async function ProjectSettingsSection() {
  await connection();
  const db = getDb();
  const project = await getActiveProject(db);
  if (!project) redirect("/proyek/baru");
  const queryClient = new QueryClient();
  queryClient.setQueryData(ledgerKeys.settings(project.id), await getProjectSettings(db, project.id));
  return (
    <HydrationBoundary state={dehydrate(queryClient)}>
      <ProjectSettingsPanel projectId={project.id} />
    </HydrationBoundary>
  );
}
```
